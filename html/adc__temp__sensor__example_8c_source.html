156static void print_temp(float temp)
157{
158 int16_t s_integer1 = 0;
159 int32_t l_integer2 = 0;
160
161 Assert(INT16_MAX > (temp * 100.0) && INT16_MIN < (temp * 100.0));
162
163 /* Cast to integer */
164 s_integer1 = (int16_t) (temp * 100.0);
165
166 /* Rounding */
167 l_integer2 = s_integer1 / 10;
168 if ((s_integer1 - l_integer2 * 10) > 4) {
169 s_integer1 = l_integer2 + 1;
170 } else {
171 if ((s_integer1 - l_integer2 * 10) < -4) {
172 s_integer1 = l_integer2 - 1;
173 } else {
174 s_integer1 = l_integer2;
175 }
176 }
177
178 /* Quotient */
179 l_integer2 = s_integer1 / 10;
180 /* Remainder */
181 s_integer1 = s_integer1 - l_integer2 * 10;
182
183 if (s_integer1 < 0) {
184 printf("Temp:-%d.%d \n\r", (int16_t) ((l_integer2) * (-1)),
185 (int16_t) ((s_integer1) * (-1)));
186 } else {
187 printf("Temp:%d.%d \n\r", (int16_t) l_integer2,
188 (int16_t) s_integer1);
189 }
190}

199static uint32_t adc_read_buffer(Adc * pADC, int16_t * pwBuffer, uint32_t dwSize)
200{
201 /* Check if the first PDC bank is free. */
202 if ((pADC->ADC_RCR == 0) && (pADC->ADC_RNCR == 0)) {
203 pADC->ADC_RPR = (uint32_t) pwBuffer;
204 pADC->ADC_RCR = dwSize;
205 pADC->ADC_PTCR = ADC_PTCR_RXTEN;
206
207 return 1;
208 } else { /* Check if the second PDC bank is free. */
209 if (pADC->ADC_RNCR == 0) {
210 pADC->ADC_RNPR = (uint32_t) pwBuffer;
211 pADC->ADC_RNCR = dwSize;
212
213 return 1;
214 } else {
215 return 0;
216 }

248void ADC_Handler(void)
249{
250 uint32_t ul_counter;
251 int32_t l_vol;
252 float f_temp;
253 uint32_t ul_value = 0;
254 uint32_t ul_temp_value = 0;
255
256 if ((adc_get_status(ADC) & ADC_ISR_RXBUFF) == ADC_ISR_RXBUFF) {
257
258 /* Multisample */
259 for (ul_counter = 0; ul_counter < BUFFER_SIZE; ul_counter++) {
260 ul_value += gs_s_adc_values[ul_counter];
261 }
262 /* Averaging */
263 ul_temp_value = ul_value / 10;
264 ul_value = ul_value / 100;
265 ul_temp_value -= (ul_value * 10);
266 /* Round for last decimal */
267 if (ul_temp_value > 4) {
268 ul_value++;
269 }
270
271 l_vol = ul_value * VOLT_REF / MAX_DIGITAL;
272#if SAM3S || SAM3XA
273 /* Using multiplication (*0.37736) instead of division (/2.65). */
274 f_temp = (float)(l_vol - 800) * 0.37736 + 27.0;
275#elif SAM4N
276 /* Using multiplication (*0.21186) instead of division (/4.72). */
277 f_temp = (float)(l_vol - 1440) * 0.21186 + 27.0;
278#else
279 /* Using multiplication (*0.21276) instead of division (/4.7). */
280 f_temp = (float)(l_vol - 1440) * 0.21276 + 27.0;
281#endif
282 print_temp(f_temp);
283 /* Clear the buffer. */
284 memset(gs_s_adc_values, 0x0, sizeof(gs_s_adc_values));
285 /* Start new pdc transfer. */
286 adc_read_buffer(ADC, gs_s_adc_values, BUFFER_SIZE);
287
288 }

305
306 configure_console();

308 /* Output example information. */
309 puts(STRING_HEADER);
310
311 /* 10 ms timer */
312 if (SysTick_Config(sysclk_get_cpu_hz() / 100)) {
313 puts("-F- Systick configuration error\r");
314 while (1) {
315 }
316 }
317 /* Enable peripheral clock. */
318 pmc_enable_periph_clk(ID_ADC);
319 /* Initialize ADC. */
320 /*
321 * Formula: ADCClock = MCK / ( (PRESCAL+1) * 2 )
322 * For example, MCK = 64MHZ, PRESCAL = 4, then:
323 * ADCClock = 64 / ((4+1) * 2) = 6.4MHz;
324 */
325 /* Formula:
326 * Startup Time = startup value / ADCClock
327 * Startup time = 64 / 6.4MHz = 10 us
328 */
329 adc_init(ADC, sysclk_get_cpu_hz(), 6400000, STARTUP_TIME);
330 /* Formula:
331 * Transfer Time = (TRANSFER * 2 + 3) / ADCClock
332 * Tracking Time = (TRACKTIM + 1) / ADCClock
333 * Settling Time = settling value / ADCClock
334 *
335 * Transfer Time = (1 * 2 + 3) / 6.4MHz = 781 ns
336 * Tracking Time = (1 + 1) / 6.4MHz = 312 ns
337 * Settling Time = 3 / 6.4MHz = 469 ns
338 */
339 adc_configure_timing(ADC, TRACKING_TIME
340#if !SAM4C && !SAM4CM
341 , ADC_SETTLING_TIME_3, TRANSFER_PERIOD
342#endif
343 );
344
345 adc_configure_trigger(ADC, ADC_TRIG_SW, 0);
346
347 adc_check(ADC, sysclk_get_cpu_hz());
348
349 /* Enable channel for potentiometer. */
350 adc_enable_channel(ADC, ADC_TEMPERATURE_SENSOR);
351
352 /* Enable the temperature sensor. */
353 adc_enable_ts(ADC);
354
355 /* Enable ADC interrupt. */
356 NVIC_EnableIRQ(ADC_IRQn);
357
358 /* Start conversion. */
359 adc_start(ADC);
360
361 adc_read_buffer(ADC, gs_s_adc_values, BUFFER_SIZE);
362
363 /* Enable PDC channel interrupt. */
364 adc_enable_interrupt(ADC, ADC_ISR_RXBUFF);
365
366 while (1) {
367 }
368}
