181void ADC_Handler(void)
182{
183 uint32_t ul_mode;
184 uint16_t us_adc;
185
186 /* Disable Compare Interrupt. */
187 adc_disable_interrupt(ADC, ADC_IDR_COMPE);
188
189 if ((adc_get_status(ADC) & ADC_ISR_COMPE) == ADC_ISR_COMPE) {
190 ul_mode = adc_get_comparison_mode(ADC);
191 us_adc = adc_get_channel_value(ADC, ADC_CHANNEL_POTENTIOMETER);
192
193 switch (ul_mode) {
194 case 0:
195 printf("-ISR-:Potentiometer voltage %d mv is below the low "
196 "threshold:%d mv!\n\r", us_adc * VOLT_REF / MAX_DIGITAL,
197 gs_us_low_threshold * VOLT_REF / MAX_DIGITAL);
198 break;
199
200 case 1:
201 printf("-ISR-:Potentiometer voltage %d mv is above the high "
202 "threshold:%d mv!\n\r", us_adc * VOLT_REF / MAX_DIGITAL,
203 gs_us_high_threshold * VOLT_REF / MAX_DIGITAL);
204 break;
205
206 case 2:
207 printf("-ISR-:Potentiometer voltage %d mv is in the comparison "
208 "window:%d-%d mv!\n\r", us_adc * VOLT_REF / MAX_DIGITAL,
209 gs_us_low_threshold * VOLT_REF / MAX_DIGITAL,
210 gs_us_high_threshold * VOLT_REF / MAX_DIGITAL);
211 break;
212
213 case 3:
214 printf("-ISR-:Potentiometer voltage %d mv is out of the comparison"
215 " window:%d-%d mv!\n\r", us_adc * VOLT_REF / MAX_DIGITAL,
216 gs_us_low_threshold * VOLT_REF / MAX_DIGITAL,
217 gs_us_high_threshold * VOLT_REF / MAX_DIGITAL);
218 break;
219 }
220 }
221}

234 tc_init(TC0, 0, 0x4 | TC_CMR_ACPC_SET | TC_CMR_WAVE
235 | TC_CMR_ACPA_CLEAR | (0x2 << 13));

238 TC0->TC_CHANNEL[0].TC_RA = 16384;
239 TC0->TC_CHANNEL[0].TC_RC = 32768;

348static int16_t get_voltage(void)
349{
350 uint8_t c_counter = 0;
351 uint8_t c_char = 0;
352 int16_t s_value = 0;
353 int8_t c_length = 0;
354 int8_t c_str_temp[5] = { 0 };
355
356 while (1) {
357 while (uart_read(CONSOLE_UART, &c_char));
358
359 if (c_char == '\n' || c_char == '\r') {
360 puts("\r");
361 break;
362 }
363
364 if ('0' <= c_char && '9' >= c_char) {
365 printf("%c", c_char);
366 c_str_temp[c_counter++] = c_char;
367#if defined ( __GNUC__ )
368 fflush(stdout);
369#endif
370
371 if (c_counter >= 4) {
372 break;
373 }
374 }
375 }
376
377 c_str_temp[c_counter] = '\0';
378 /* Input string length. */
379 c_length = c_counter;
380 s_value = 0;
381
382 /* Convert string to integer. */
383 for (c_counter = 0; c_counter < 4; c_counter++) {
384 if (c_str_temp[c_counter] != '0') {
385 switch (c_length - c_counter - 1) {
386 case 0:
387 s_value += (c_str_temp[c_counter] - '0');
388 break;
389
390 case 1:
391 s_value += (c_str_temp[c_counter] - '0') * 10;
392 break;
393
394 case 2:
395 s_value += (c_str_temp[c_counter] - '0') * 100;
396 break;
397
398 case 3:
399 s_value += (c_str_temp[c_counter] - '0') * 1000;
400 break;
401 }
402 }
403 }
404
405 if (s_value > VOLT_REF) {
406 puts("\n\r-F- Too big threshold!\r");
407 return -1;
408 }
409
410 return s_value;
411}

433int main(void)
434{
435 uint8_t c_choice;
436 int16_t s_adc_value;
437 int16_t s_threshold = 0;
438
439 /* Initialize the SAM system. */
440 sysclk_init();
441 board_init();
442
443 configure_console();
444
445 /* Output example information. */
446 puts(STRING_HEADER);
447
448 /* Initialize threshold. */
449 gs_us_low_threshold = 0x0;
450 gs_us_high_threshold = MAX_DIGITAL;
451
452 /* Enable peripheral clock. */
453 pmc_enable_periph_clk(ID_ADC);
454
455 /* Initialize ADC. */
456 /*
457 * Formula: ADCClock = MCK / ( (PRESCAL+1) * 2 )
458 * For example, MCK = 64MHZ, PRESCAL = 4, then:
459 * ADCClock = 64 / ((4+1) * 2) = 6.4MHz;
460 */
461 /* Formula:
462 * Startup Time = startup value / ADCClock
463 * Startup time = 64 / 6.4MHz = 10 us
464 */
465 adc_init(ADC, sysclk_get_cpu_hz(), 6400000, STARTUP_TIME);
466
467 /* Formula:
468 * Transfer Time = (TRANSFER * 2 + 3) / ADCClock
469 * Tracking Time = (TRACKTIM + 1) / ADCClock
470 * Settling Time = settling value / ADCClock
471 *
472 * Transfer Time = (1 * 2 + 3) / 6.4MHz = 781 ns
473 * Tracking Time = (1 + 1) / 6.4MHz = 312 ns
474 * Settling Time = 3 / 6.4MHz = 469 ns
475 */
476#if SAM3S || SAM3XA || SAM4S
477 adc_configure_timing(ADC, TRACKING_TIME, ADC_SETTLING_TIME_3,
478 TRANSFER_PERIOD);
479#elif SAM3N || SAM4C
480 adc_configure_timing(ADC, TRACKING_TIME);
481#endif
482 adc_check(ADC, sysclk_get_cpu_hz());
483
484 /* Hardware trigger TIOA0. */
485 adc_configure_trigger(ADC, ADC_TRIG_TIO_CH_0, 0);
486 /* Enable channels for x,y and z. */
487 adc_enable_channel(ADC, ADC_CHANNEL_POTENTIOMETER);
488
489 /* Configure TC. */
490 configure_tc0();
491
492 /* Channel 5 has to be compared. */
493 adc_set_comparison_channel(ADC, ADC_CHANNEL_POTENTIOMETER);
494 /* Compare mode, in the window. */
495 adc_set_comparison_mode(ADC, ADC_EMR_CMPMODE_IN);
496
497 /* Set up Threshold. */
498 adc_set_comparison_window(ADC, gs_us_low_threshold, gs_us_high_threshold);
499
500 /* Enable ADC interrupt. */
501 NVIC_EnableIRQ(ADC_IRQn);
502
503 /* Start TC0 and hardware trigger. */
504 tc_start(TC0, 0);
505
506 /* Display main menu. */
507 display_menu();
508
509 while (1) {
510 while (uart_read(CONSOLE_UART, &c_choice)) {
511 }
512 printf("%c\r\n", c_choice);
513
514 switch (c_choice) {
515 case '0':
516 s_adc_value = adc_get_channel_value(ADC,
517 ADC_CHANNEL_POTENTIOMETER);
518 printf("-I- Current voltage is %d mv, %d%% of ADVREF\n\r",
519 (s_adc_value * VOLT_REF / MAX_DIGITAL),
520 (s_adc_value * 100 / MAX_DIGITAL));
521 break;
522
523 case '1':
524 puts("Low threshold is set to(mv):");
525 s_threshold = get_voltage();
526 puts("\r");
527
528 if (s_threshold >= 0) {
529 s_adc_value = s_threshold * MAX_DIGITAL /
530 VOLT_REF;
531 adc_set_comparison_window(ADC, s_adc_value,
532 gs_us_high_threshold);
533 /* Renew low threshold. */
534 gs_us_low_threshold = s_adc_value;
535 float f_low_threshold = (float)gs_us_low_threshold *
536 VOLT_REF / MAX_DIGITAL;
537 uint32_t ul_low_threshold = f_to_int(f_low_threshold);
538 printf("Setting low threshold to %u mv (reg value to 0x%x ~%d%%)\n\r",
539 (unsigned int)ul_low_threshold,
540 gs_us_low_threshold,
541 gs_us_low_threshold * 100 / MAX_DIGITAL);
542 }
543 break;
544
545 case '2':
546 puts("High threshold is set to(mv):");
547 s_threshold = get_voltage();
548 puts("\r");
549
550 if (s_threshold >= 0) {
551 s_adc_value = s_threshold * MAX_DIGITAL /
552 VOLT_REF;
553 adc_set_comparison_window(ADC, gs_us_low_threshold,
554 s_adc_value);
555 /* Renew high threshold. */
556 gs_us_high_threshold = s_adc_value;
557 float f_high_threshold = (float)gs_us_high_threshold *
558 VOLT_REF / MAX_DIGITAL;
559 uint32_t ul_high_threshold = f_to_int(f_high_threshold);
560 printf("Setting high threshold to %u mv (reg value to 0x%x ~%d%%)\n\r",
561 (unsigned int)ul_high_threshold,
562 gs_us_high_threshold,
563 gs_us_high_threshold * 100 / MAX_DIGITAL);
564 }
565 break;
566 case '3':
567 puts("-a. Below low threshold.\n\r"
568 "-b. Above high threshold.\n\r"
569 "-c. In the comparison window.\n\r"
570 "-d. Out of the comparison window.\n\r"
571 "-q. Quit the setting.\r");
572 c_choice = get_comparison_mode();
573 adc_set_comparison_mode(ADC, c_choice);
574 printf("Comparison mode is %c.\n\r", 'a' + c_choice);
575 break;
576
577 case 'm':
578 case 'M':
579 display_menu();
580 break;
581
582 case 'i':
583 case 'I':
584 display_info();
585 break;
586
587 case 's':
588 case 'S':
589 enter_asleep();
590 break;
591 }
592 puts("Press \'m\' or \'M\' to display the main menu again!\r");
593 }
