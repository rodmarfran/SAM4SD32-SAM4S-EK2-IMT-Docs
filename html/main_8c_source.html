184static void Int1Handler(uint32_t ul_id, uint32_t ul_mask)
185{
186 if (PIN_PUSHBUTTON_1_ID != ul_id || PIN_PUSHBUTTON_1_MASK != ul_mask)
187 return;
188
189 pio_disable_interrupt(PIN_PUSHBUTTON_1_PIO, PIN_PUSHBUTTON_1_MASK);
190
191 /* Trigger INT1 & Delay for a while. */
192 puts("===================================================\r");
193 puts("Enter _Int1Handler.\r");
194
195 gpio_set_pin_low(LED0_GPIO);
196
197 delay_ticks(2000);
198
199 gpio_set_pin_high(LED0_GPIO);
200
201 puts("Exit _Int1Handler.\r");
202 puts("===================================================\r");
203
204 pio_enable_interrupt(PIN_PUSHBUTTON_1_PIO, PIN_PUSHBUTTON_1_MASK);
205}

247static void set_interrupt_priority(uint8_t int1Prior, uint8_t int2Prior)
248{
249 /* Set INT1 priority. */
250 NVIC_DisableIRQ((IRQn_Type) PIN_PUSHBUTTON_1_ID);
251 NVIC_ClearPendingIRQ((IRQn_Type) PIN_PUSHBUTTON_1_ID);
252 NVIC_SetPriority((IRQn_Type) PIN_PUSHBUTTON_1_ID, int1Prior);
253 NVIC_EnableIRQ((IRQn_Type) PIN_PUSHBUTTON_1_ID);

255 /* Set INT2 priority. */

257 NVIC_ClearPendingIRQ((IRQn_Type) PIN_PUSHBUTTON_2_ID);
258 NVIC_SetPriority((IRQn_Type) PIN_PUSHBUTTON_2_ID, int2Prior);
259 NVIC_EnableIRQ((IRQn_Type) PIN_PUSHBUTTON_2_ID);

268static void configure_buttons(void)
269{
270 /* Adjust PIO debouncing filter patameters, using 10 Hz filter. */
271 pio_set_debounce_filter(PIN_PUSHBUTTON_1_PIO, PIN_PUSHBUTTON_1_MASK, 10);
272 pio_set_debounce_filter(PIN_PUSHBUTTON_2_PIO, PIN_PUSHBUTTON_2_MASK, 10);
273
274 /* Initialize PIOs interrupt handlers (see PIO definition in board.h). */
275 /* Interrupt on rising edge. */
276 pio_handler_set(PIN_PUSHBUTTON_1_PIO, PIN_PUSHBUTTON_1_ID,
277 PIN_PUSHBUTTON_1_MASK, PIN_PUSHBUTTON_1_ATTR, Int1Handler);
278 /* Interrupt on falling edge. */
279 pio_handler_set(PIN_PUSHBUTTON_2_PIO, PIN_PUSHBUTTON_2_ID,
280 PIN_PUSHBUTTON_2_MASK, PIN_PUSHBUTTON_2_ATTR, Int2Handler);
281
282 /* Enable PIO controller IRQs. */
283 NVIC_EnableIRQ((IRQn_Type) PIN_PUSHBUTTON_1_ID);
284 NVIC_EnableIRQ((IRQn_Type) PIN_PUSHBUTTON_2_ID);
285
286 /* Enable PIO line interrupts. */
287 pio_enable_interrupt(PIN_PUSHBUTTON_1_PIO, PIN_PUSHBUTTON_1_MASK);
288 pio_enable_interrupt(PIN_PUSHBUTTON_2_PIO, PIN_PUSHBUTTON_2_MASK);
289}

326int main(void)
327{
328 uint8_t uc_key;
329
330 /* Initialize the SAM4 system */
331 sysclk_init();
332 board_init();
333
334 WDT->WDT_MR = WDT_MR_WDDIS;
335
336 /* Enable the pmc clocks of the push buttons for all SAM4. */
337 pmc_enable_periph_clk(ID_PIOA);
338 pmc_enable_periph_clk(ID_PIOB);
339 pmc_enable_periph_clk(ID_PIOC);
340
341 /* Initialize the console uart */
342 configure_console();
343
344 /* Output example information */
345 puts(STRING_HEADER);
346
347 /* PIO configuration for LEDs and Buttons. */
348 pio_handler_set_priority(PIOA, PIOA_IRQn, 0);
349 pio_handler_set_priority(PIOB, PIOB_IRQn, 0);
350 pio_handler_set_priority(PIOC, PIOC_IRQn, 0);
351
352 /* configure LED. */
353 led_config();
354
355 /* configure push buttons. */
356 configure_buttons();
357
358 /* Set default priorities for 2 buttons. */
359 puts("Set INT1's priority higher than INT2.\r");
360 set_interrupt_priority(INT_PRIOR_HIGH, INT_PRIOR_LOW);
361
362 /* Display the main menu. */
363 display_menu();
364
365 // Flash the LED.
366 while (1) {
367 while (uart_read(CONSOLE_UART, &uc_key));
368
369 switch (uc_key) {
370 case '1':
371 set_interrupt_priority(INT_PRIOR_LOW, INT_PRIOR_HIGH);
372 puts("Set INT2's priority higher than INT1.\n\r\r");
373 break;
374
375 case '2':
376 set_interrupt_priority(INT_PRIOR_HIGH, INT_PRIOR_LOW);
377 puts("Set INT1's priority higher than INT2.\n\r\r");
378 break;
379
380 case 'h':
381 display_menu();
382 break;
383
384 default:
385 puts("Invalid input.\r");
386 break;
387 }
388 }
389}
