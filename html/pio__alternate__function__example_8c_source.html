125static void configure_button(void)
126{
127 /* Configure PIO clock. */
128 pmc_enable_periph_clk(PUSH_BUTTON_ID);
129
130 /* Adjust pio debounce filter parameters, uses 10 Hz filter. */
131 pio_set_debounce_filter(PUSH_BUTTON_PIO, PUSH_BUTTON_PIN_MSK, 10);
132
133 /* Initialize pios interrupt handlers, see PIO definition in board.h. */
134 pio_handler_set(PUSH_BUTTON_PIO, PUSH_BUTTON_ID, PUSH_BUTTON_PIN_MSK,
135 PUSH_BUTTON_ATTR, button_handler);
136
137 /* Enable PIO controller IRQs. */
138 NVIC_EnableIRQ((IRQn_Type)PUSH_BUTTON_ID);
139
140 /* Enable PIO line interrupts. */
141 pio_enable_interrupt(PUSH_BUTTON_PIO, PUSH_BUTTON_PIN_MSK);
142}

187 sysclk_init();

193 /* Initialize flash: 6 wait states for flash writing. */
194 flash_init(FLASH_ACCESS_MODE_128, FLASH_WAIT_STATE_NBR);
195
196 /* Configure Push Button. */
197 configure_button();
198
199 /* Output example information. */
200 puts(STRING_HEADER);
201
202 puts("-I- Unlocking the whole flash.\r\n");
203 /* Unlock the whole flash. */
204 ul_error = flash_unlock(IFLASH_ADDR, (IFLASH_ADDR + IFLASH_SIZE - 1), 0, 0);
205 if (FLASH_RC_OK != ul_error) {
206 puts("Unlock internal flash failed.\r\n");
207 return 0;
208 }
209
210 /* Perform tests on the test page. */
211 p_test_page_data = (volatile uint32_t *)TEST_PAGE_ADDRESS;
212
213 /* Write page with walking bit pattern (0x00000001, 0x00000002, ...). */
214 puts("-I- Writing test page with walking bit pattern.\r\n");
215 for (i = 0; i < BUFFER_SIZE; i++) {
216 p_buffer[i] = 1 << (i % MAX_SHIFTING_NUMBER);
217 }
218#if (SAM4E || SAM4C || SAME70 || SAMV71)
223 ul_error = flash_erase_sector(TEST_PAGE_ADDRESS);
224 if (ul_error != FLASH_RC_OK) {
225 printf("-F- Flash erase error %u\n\r", ul_error);
226 return 0;
227 }
228
229 ul_error = flash_write(TEST_PAGE_ADDRESS, p_buffer,
230 IFLASH_PAGE_SIZE, 0);
231#else
232 ul_error = flash_write(TEST_PAGE_ADDRESS, p_buffer,
233 IFLASH_PAGE_SIZE, 1);
234#endif
235 if (FLASH_RC_OK != ul_error) {
236 puts("Write the test page of internal flash failed.\r\n");
237 return 0;
238 }
239
240 /* Check page contents. */
241 puts("-I- Checking page contents.\r\n");
242 for (i = 0; i < BUFFER_SIZE; i++) {
243 printf(".");
244 if (p_test_page_data[i] != (1u << (i % MAX_SHIFTING_NUMBER))) {
245 puts("The content in the test page isn't written correctly");
246 return 0;
247 }
248 }
249 puts(" OK! \r\n");
250
251 /* Configure Erase pin NOT in Erase mode. */
252 puts("-I- Configure Erase pin in PIO mode.\r\n");
253 matrix_set_system_io(PIN_PIO_MODE_MSK);
254
258 printf("-I- Please close the erase jumper and then open it ");
259 printf("at least 200ms later.\r\n");
260
261 printf("Then press button %s to go on!\r\n", BUTTON_STRING);
262 /* Wait until Push Button is pressed. */
263 while (!g_button_event) {
264 }
269 pio_disable_interrupt(PUSH_BUTTON_PIO, PUSH_BUTTON_PIN_MSK);
270 g_button_event = 0;
271 /* Read the page again, it should be unchanged. */
272 puts("-I- Reading the page\r\n");
273 for (i = 0; i < BUFFER_SIZE; i++) {
274 printf(".");
275 if (p_test_page_data[i] != (1u << (i % MAX_SHIFTING_NUMBER))) {
276 puts("-F- Reading Error! \r\n");
277 return 0;
278 }
279 }
280 puts("\r\n");
281 puts("Read OK! Erase is out of function!\r\n");
282 /* Configure Erase pin as Erase function. */
283 puts("-I- Configure Erase pin as Erase function\r\n");
284 matrix_set_system_io(PIN_ERASE_MODE_MSK);
285
289 printf("-I- Please close the erase jumper and then open it ");
290 printf("at least 200ms later.\r\n");
291
296 printf("-I- As the internal flash has been erased and the code can't ");
297 printf("be executed any more, users can press the reset button on EK ");
298 printf("and will see there will be no output message any more.\r\n");
299
300 while (1) {
301 }
302}
