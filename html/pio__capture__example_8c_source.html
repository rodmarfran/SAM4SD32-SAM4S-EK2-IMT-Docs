183static void capture_handler(Pio *p_pio)
184{
185 uint8_t uc_i;
186 uint32_t dummy_data;
187
188 pio_capture_disable_interrupt(p_pio, (PIO_PCIDR_ENDRX | PIO_PCIDR_RXBUFF));
189 /* Disable PDC transfer. */
190 pdc_disable_transfer(p_pdc, PERIPH_PTCR_RXTEN);
191 /* Clear any unwanted data */
192 pio_capture_read(PIOA, &dummy_data);
193
194 printf("End of receive.\r\n");
195 for (uc_i = 0; uc_i < SIZE_BUFF_RECEPT; uc_i++) {
196 printf("0x%X ", pio_rx_buffer[uc_i]);
197 }
198 printf("\r\n");
199 g_uc_cbk_received = 1;
200}

233 sysclk_init();
234 board_init();

237 configure_console();
238
239 /* Configure PIOA clock. */
240 pmc_enable_periph_clk(ID_PIOA);
241 pmc_enable_periph_clk(ID_PIOC);
242
243 /* Configure PIO Capture handler */
244 pio_capture_handler_set(capture_handler);
245
246 /* Output example information. */
247 puts(STRING_HEADER);
248
249 printf("Frequency: %d MHz.\r\n",
250 (uint8_t) (sysclk_get_cpu_hz() / 1000000));
251
252 printf("Press r to Receive data on PIO Parallel Capture.\r\n");
253 printf("Press s to Send data on PIO Parallel Capture.\r\n");
254 uc_key = 0;
255 while ((uc_key != 'r') && (uc_key != 's')) {
256 uart_read(CONSOLE_UART, &uc_key);
257 }
258 if (uc_key == 'r') {
259 printf("** RECEIVE mode **\r\n");
260
261 /* Initialize PIO capture mode value. */
262 ul_mode = 0;
263 /* Set up the parallel capture mode data size as 8 bits. */
264 ul_mode |= 0 << PIO_PCMR_DSIZE_Pos;
265
266 printf("Press y to sample the data when both data enable pins are enabled.\r\n");
267 printf("Press n to sample the data, don't care the status of the data enable pins.\r\n");
268 uc_key = 0;
269 while ((uc_key != 'y') && (uc_key != 'n')) {
270 uart_read(CONSOLE_UART, &uc_key);
271 }
272 if (uc_key == 'y') {
273 /* Sample the data when both data enable pins are enabled. */
274 ul_mode &= ~PIO_PCMR_ALWYS;
275 printf("Receive data when both data enable pins are enabled.\r\n");
276 } else {
277 /* Sample the data, don't care the status of the data enable pins. */
278 ul_mode |= PIO_PCMR_ALWYS;
279 printf("Receive data, don't care the status of the data enable pins.\r\n");
280 }
281 printf("Press y to sample all the data\r\n");
282 printf("Press n to sample the data only one out of two.\r\n");
283 uc_key = 0;
284 while ((uc_key != 'y') && (uc_key != 'n')) {
285 uart_read(CONSOLE_UART, &uc_key);
286 }
287 if (uc_key == 'y') {
288 /* Sample all the data. */
289 ul_mode &= ~PIO_PCMR_HALFS;
290 printf("All data are sampled.\r\n");
291 } else {
292 /* Sample the data only one out of two. */
293 ul_mode |= PIO_PCMR_HALFS;
294 /* Only if half-Sampling is set, data with an even index are sampled. */
295 ul_mode &= ~PIO_PCMR_FRSTS;
296 printf("Only one out of two data is sampled, with an even index.\r\n");
297 }
298
299 /* Initialize PIO Parallel Capture function. */
300 pio_capture_set_mode(PIOA, ul_mode);
301 pio_capture_enable(PIOA);
302
303 /* Disable all PIOA I/O line interrupt. */
304 pio_disable_interrupt(PIOA, 0xFFFFFFFF);
305
306 /* Configure and enable interrupt of PIO. */
307 NVIC_DisableIRQ(PIOA_IRQn);
308 NVIC_ClearPendingIRQ(PIOA_IRQn);
309 NVIC_SetPriority(PIOA_IRQn, PIO_IRQ_PRI);
310 NVIC_EnableIRQ(PIOA_IRQn);
311
312 while (1) {
313 g_uc_cbk_received = 0;
314
315 /* Clear Receive buffer. */
316 for (uc_i = 0; uc_i < SIZE_BUFF_RECEPT; uc_i++) {
317 pio_rx_buffer[uc_i] = 0;
318 }
319
320 /* Set up PDC receive buffer, waiting for 64 bytes. */
321 packet_t.ul_addr = (uint32_t) pio_rx_buffer;
322 packet_t.ul_size = SIZE_BUFF_RECEPT;
323 p_pdc = pio_capture_get_pdc_base(PIOA);
324 pdc_rx_init(p_pdc, &packet_t, NULL);
325
326 /* Enable PDC transfer. */
327 pdc_enable_transfer(p_pdc, PERIPH_PTCR_RXTEN);
328
329 /* Configure the PIO capture interrupt mask. */
330 pio_capture_enable_interrupt(PIOA,
331 (PIO_PCIER_ENDRX | PIO_PCIER_RXBUFF));
332
333 printf("Waiting...\r\n");
334 while (g_uc_cbk_received == 0) {
335 }
336 }
337 } else if (uc_key == 's') {
338 printf("** SEND mode **\r\n");
339 printf("This is for debug purpose only !\r\n");
340 printf("Frequency of PIO controller clock must be strictly superior");
341 printf("to 2 times the frequency of the clock of the device which");
342 printf(" generates the parallel data.\r\n");
343 printf("\r\nPlease connect the second board, ");
344 printf("and put it in receive mode.\r\n");
345
346 /* Configure PIO pins which simulate as a sensor. */
347 pio_configure_pin_group(PIOA, PIO_CAPTURE_CONTROL_PIN_MSK,
348 PIO_CAPTURE_OUTPUT_PIN_FLAGS);
349 pio_configure_pin_group(PIOC, PIO_CAPTURE_DATA_PINS_MASK,
350 PIO_CAPTURE_OUTPUT_PIN_FLAGS);
351 pio_set_pin_low(PIO_CAPTURE_EN1_IDX);
352 pio_set_pin_low(PIO_CAPTURE_EN2_IDX);
353 pio_set_pin_low(PIO_CAPTURE_CCLK_IDX);
354
355 /* Enable sync. output data. */
356 pio_enable_output_write(PIOC, PIO_CAPTURE_DATA_PINS_MASK);
357
358 /* Initialize the capture data line. */
359 pio_sync_output_write(PIOC, 0);
360
361 printf("Press y to send data with data enable pins.\r\n");
362 printf("Press n to send data without data enable pins.\r\n");
363
364 uc_key = 0;
365 while ((uc_key != 'y') && (uc_key != 'n')) {
366 uart_read(CONSOLE_UART, &uc_key);
367 }
368 if (uc_key == 'y') {
369 uc_tx_without_en = 0;
370 printf("Send data with both data enable pins enabled.\r\n");
371 } else {
372 uc_tx_without_en = 1;
373 printf("Send data without enabling the data enable pins.\r\n");
374 }
375
376 printf("Press y to indicate that receiver samples all data.\r\n");
377 printf("Press n to indicate that receiver samples data with an even index.\r\n");
378 uc_key = 0;
379 while ((uc_key != 'y') && (uc_key != 'n')) {
380 uart_read(CONSOLE_UART, &uc_key);
381 }
382 if (uc_key == 'y') {
383 uc_rx_even_only = 0;
384 printf("Receiver samples all data.\r\n");
385 } else {
386 uc_rx_even_only = 1;
387 printf("Receiver samples data with an even index.\r\n");
388 }
389
390 ul_length = SIZE_BUFF_RECEPT * (1 + uc_rx_even_only);
391 while (1) {
392 if (uc_tx_without_en) {
393 printf("\r\nSend data without enabling the data enable pins.\r\n");
394 } else {
395 printf("\r\nSend data with both data enable pins enabled.\r\n");
396 }
397 if (!uc_tx_without_en) {
398 /* Set enable pins. */
399 pio_set_pin_high(PIO_CAPTURE_EN1_IDX);
400 pio_set_pin_high(PIO_CAPTURE_EN2_IDX);
401 }
402 for (uc_i = 0; uc_i < ul_length;) {
403 /* Send data. */
404 pio_sync_output_write(PIOC,
405 (uc_i << PIO_CAPTURE_DATA_POS));
406 /* Set clock. */
407 pio_set_pin_high(PIO_CAPTURE_CCLK_IDX);
408 delay_us(20);
409 /* Clear clock. */
410 pio_set_pin_low(PIO_CAPTURE_CCLK_IDX);
411 delay_us(20);
412 uc_i++;
413 }
414 if (!uc_tx_without_en) {
415 /* Clear enable pins. */
416 pio_set_pin_low(PIO_CAPTURE_EN1_IDX);
417 pio_set_pin_low(PIO_CAPTURE_EN2_IDX);
418 }
419 printf("Press a key.\r\n");
420 while (uart_read(CONSOLE_UART, &uc_key)) {
421 }
422 }
423 }
424
425 return 0;
426}
