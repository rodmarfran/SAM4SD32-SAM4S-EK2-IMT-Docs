256static void spi_slave_command_process(void)
257{
258 if (gs_ul_spi_cmd == CMD_END) {
259 gs_ul_spi_state = SLAVE_STATE_IDLE;
260 gs_spi_status.ul_total_block_number = 0;
261 gs_spi_status.ul_total_command_number = 0;
262 } else {
263 switch (gs_ul_spi_state) {
264 case SLAVE_STATE_IDLE:
265 /* Only CMD_TEST accepted. */
266 if (gs_ul_spi_cmd == CMD_TEST) {
267 gs_ul_spi_state = SLAVE_STATE_TEST;
268 }
269 break;
270
271 case SLAVE_STATE_TEST:
272 /* Only CMD_DATA accepted. */
273 if ((gs_ul_spi_cmd & CMD_DATA_MSK) == CMD_DATA) {
274 gs_ul_spi_state = SLAVE_STATE_DATA;
275 }
276 gs_ul_test_block_number = gs_ul_spi_cmd & DATA_BLOCK_MSK;
277 break;
278
279 case SLAVE_STATE_DATA:
280 gs_spi_status.ul_total_block_number++;
281
282 if (gs_spi_status.ul_total_block_number ==
283 gs_ul_test_block_number) {
284 gs_ul_spi_state = SLAVE_STATE_STATUS_ENTRY;
285 }
286 break;
287
288 case SLAVE_STATE_STATUS_ENTRY:
289 gs_ul_spi_state = SLAVE_STATE_STATUS;
290 break;
291
292 case SLAVE_STATE_END:
293 break;
294 }
295 }

301static void spi_slave_new_command(void)
302{
303 switch (gs_ul_spi_state) {
304 case SLAVE_STATE_IDLE:
305 case SLAVE_STATE_END:
306 gs_ul_spi_cmd = RC_SYN;
307 spi_slave_transfer(&gs_ul_spi_cmd, sizeof(gs_ul_spi_cmd));
308 break;
309
310 case SLAVE_STATE_TEST:
311 gs_ul_spi_cmd = RC_RDY;
312 spi_slave_transfer(&gs_ul_spi_cmd, sizeof(gs_ul_spi_cmd));
313 break;
314
315 case SLAVE_STATE_DATA:
316 if (gs_spi_status.ul_total_block_number < gs_ul_test_block_number) {
317 spi_slave_transfer(gs_uc_spi_buffer, COMM_BUFFER_SIZE);
318 }
319 break;
320
321 case SLAVE_STATE_STATUS_ENTRY:
322 gs_ul_spi_cmd = RC_RDY;
323 spi_slave_transfer(&gs_ul_spi_cmd, sizeof(gs_ul_spi_cmd));
324 gs_ul_spi_state = SLAVE_STATE_STATUS;
325 break;
326
327 case SLAVE_STATE_STATUS:
328 gs_ul_spi_cmd = RC_SYN;
329 spi_slave_transfer(&gs_spi_status, sizeof(struct status_block_t));
330 gs_ul_spi_state = SLAVE_STATE_END;
331 break;
332 }

338void SPI_Handler(void)
339{
340 uint32_t new_cmd = 0;
341 static uint16_t data;
342 uint8_t uc_pcs;
343
344 if (spi_read_status(SPI_SLAVE_BASE) & SPI_SR_RDRF) {
345 spi_read(SPI_SLAVE_BASE, &data, &uc_pcs);
346 gs_puc_transfer_buffer[gs_ul_transfer_index] = data;
347 gs_ul_transfer_index++;
348 gs_ul_transfer_length--;
349 if (gs_ul_transfer_length) {
350 spi_write(SPI_SLAVE_BASE,
351 gs_puc_transfer_buffer[gs_ul_transfer_index], 0, 0);
352 }
353
354 if (!gs_ul_transfer_length) {
355 spi_slave_command_process();
356 new_cmd = 1;
357 }
358
359 if (new_cmd) {
360 if (gs_ul_spi_cmd != CMD_END) {
361 gs_spi_status.ul_cmd_list[gs_spi_status.ul_total_command_number]
362 = gs_ul_spi_cmd;
363 gs_spi_status.ul_total_command_number++;
364 }
365 spi_slave_new_command();
366 }
367 }

466static void spi_master_transfer(void *p_buf, uint32_t size)
467{
468 uint32_t i;
469 uint8_t uc_pcs;
470 static uint16_t data;
471
472 uint8_t *p_buffer;
473
474 p_buffer = p_buf;
475
476 for (i = 0; i < size; i++) {
477 spi_write(SPI_MASTER_BASE, p_buffer[i], 0, 0);
478 /* Wait transfer done. */
479 while ((spi_read_status(SPI_MASTER_BASE) & SPI_SR_RDRF) == 0);
480 spi_read(SPI_MASTER_BASE, &data, &uc_pcs);
481 p_buffer[i] = data;

488static void spi_master_go(void)
489{
490 uint32_t cmd;
491 uint32_t block;
492 uint32_t i;
493
494 /* Configure SPI as master, set up SPI clock. */
495 spi_master_initialize();
496
497 /*
498 * Send CMD_TEST to indicate the start of test, and device shall return
499 * RC_RDY.
500 */
501 while (1) {
502 cmd = CMD_TEST;
503 puts("-> Master sending CMD_TEST... \r");
504 spi_master_transfer(&cmd, sizeof(cmd));
505 if (cmd == RC_RDY) {
506 puts(" <- Slave response RC_SYN, RC_RDY \r");
507 break;
508 }
509 if (cmd != RC_SYN) {
510 printf("-E- Response unexpected: 0x%x \n\r", (unsigned)cmd);
511 return;
512 }
513 }
514 /* Send CMD_DATA with 4 blocks (64 bytes per page). */
515 puts("-> Master sending CMD_DATA... \r");
516 cmd = CMD_DATA | MAX_DATA_BLOCK_NUMBER;
517 spi_master_transfer(&cmd, sizeof(cmd));
518 puts(" <---- Slave response RC_RDY \r");
519 for (block = 0; block < MAX_DATA_BLOCK_NUMBER; block++) {
520 for (i = 0; i < COMM_BUFFER_SIZE; i++) {
521 gs_uc_spi_buffer[i] = block;
522 }
523 printf("-> Master sending block %u ... \n\r", (unsigned)block);
524 spi_master_transfer(gs_uc_spi_buffer, COMM_BUFFER_SIZE);
525 if (block) {
526 for (i = 0; i < COMM_BUFFER_SIZE; i++) {
527 if (gs_uc_spi_buffer[i] != (block - 1)) {
528 break;
529 }
530 }
531 if (i < COMM_BUFFER_SIZE) {
532 printf("-E- block %u contains unexpected data \n\r",
533 (unsigned)block);
534 } else {
535 printf(" <- Slave response last block %x \n\r",
536 (unsigned)(block - 1));
537 }
538 }
539 }
540
541 for (i = 0; i < MAX_RETRY; i++) {
542 cmd = CMD_STATUS;
543 puts("-> Master sending CMD_STATUS... \r");
544 spi_master_transfer(&cmd, sizeof(cmd));
545 if (cmd == RC_RDY) {
546 puts(" <- Slave response RC_RDY \r");
547 break;
548 }
549 }
550 if (i >= MAX_RETRY) {
551 puts(" <- Slave no response \r");
552 return;
553 }
554
555 puts("-> Master request slave status... \r");
556 spi_master_transfer(&gs_spi_status, sizeof(struct status_block_t));
557
558 puts(" <- Slave reports status...\r");
559 printf("-I- Received %u commands:",
560 (unsigned)gs_spi_status.ul_total_command_number);
561
562 for (i = 0; i < gs_spi_status.ul_total_command_number; i++) {
563 printf(" 0x%08x", (unsigned)gs_spi_status.ul_cmd_list[i]);
564 }
565 printf(" \n\r-I- Received %lu data blocks \n\r",
566 (unsigned long)gs_spi_status.ul_total_block_number);
567
568 for (i = 0; i < MAX_RETRY; i++) {
569 puts("-> Master sending CMD_END... \r");
570 cmd = CMD_END;
571 spi_master_transfer(&cmd, sizeof(cmd));
572
573 if (cmd == RC_SYN) {
574 puts(" <- Slave response RC_SYN \r");
575 break;
576 }
577 }
578
579 if (i >= MAX_RETRY) {
580 puts(" <- Slave no response \r");
581 }
582
583 puts("\r");
584 puts("SPI transfer test finished! \r");
585}
