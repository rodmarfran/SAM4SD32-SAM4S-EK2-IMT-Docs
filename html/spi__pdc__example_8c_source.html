185static void spi_slave_transfer(void *p_tbuf, uint32_t tsize, void *p_rbuf,
186 uint32_t rsize)
187{
188 uint32_t spi_ier;
189 pdc_packet_t pdc_spi_packet;
190
191 pdc_spi_packet.ul_addr = (uint32_t)p_rbuf;
192 pdc_spi_packet.ul_size = rsize;
193 pdc_rx_init(g_p_spis_pdc, &pdc_spi_packet, NULL);
194
195 pdc_spi_packet.ul_addr = (uint32_t)p_tbuf;
196 pdc_spi_packet.ul_size = tsize;
197 pdc_tx_init(g_p_spis_pdc, &pdc_spi_packet, NULL);
198
199 /* Enable the RX and TX PDC transfer requests */
200 pdc_enable_transfer(g_p_spis_pdc, PERIPH_PTCR_RXTEN | PERIPH_PTCR_TXTEN);
201
202 /* Transfer done handler is in ISR */
203 spi_ier = SPI_IER_NSSR | SPI_IER_RXBUFF;
204 spi_enable_interrupt(SPI_SLAVE_BASE, spi_ier) ;
205}

432 break;
433

435 spi_master_go();
436 break;
437

440 if ((uc_key >= '0')
441 && (uc_key <= ('0' + NUM_SPCK_CONFIGURATIONS - 1))) {
442 spi_set_clock_configuration(uc_key - '0');
443 }
444 break;
445 }
446 }
