120static void tc_waveform_initialize(void)
121{
122 uint32_t ra, rc;
123
124 /* Configure the PMC to enable the TC module. */
125 sysclk_enable_peripheral_clock(ID_TC_WAVEFORM);
126#if SAMG55
127 /* Enable PCK output */
128 pmc_disable_pck(PMC_PCK_3);
129 pmc_switch_pck_to_mck(PMC_PCK_3, PMC_PCK_PRES(0));
130 pmc_enable_pck(PMC_PCK_3);
131#endif
132
133 /* Init TC to waveform mode. */
134 tc_init(TC, TC_CHANNEL_WAVEFORM,
135 /* Waveform Clock Selection */
136 gc_waveconfig[gs_uc_configuration].ul_intclock
137 | TC_CMR_WAVE /* Waveform mode is enabled */
138 | TC_CMR_ACPA_SET /* RA Compare Effect: set */
139 | TC_CMR_ACPC_CLEAR /* RC Compare Effect: clear */
140 | TC_CMR_CPCTRG /* UP mode with automatic trigger on RC Compare */
141 );
142
143 /* Configure waveform frequency and duty cycle. */
144 rc = (sysclk_get_peripheral_bus_hz(TC) /
145 divisors[gc_waveconfig[gs_uc_configuration].ul_intclock]) /
146 gc_waveconfig[gs_uc_configuration].us_frequency;
147 tc_write_rc(TC, TC_CHANNEL_WAVEFORM, rc);
148 ra = (100 - gc_waveconfig[gs_uc_configuration].us_dutycycle) * rc / 100;
149 tc_write_ra(TC, TC_CHANNEL_WAVEFORM, ra);
150
151 /* Enable TC TC_CHANNEL_WAVEFORM. */
152 tc_start(TC, TC_CHANNEL_WAVEFORM);
153 printf("Start waveform: Frequency = %d Hz,Duty Cycle = %2d%%\n\r",
154 gc_waveconfig[gs_uc_configuration].us_frequency,
155 gc_waveconfig[gs_uc_configuration].us_dutycycle);
156}

162static void tc_capture_initialize(void)
163{
164 /* Configure the PMC to enable the TC module */
165 sysclk_enable_peripheral_clock(ID_TC_CAPTURE);
166#if SAMG55
167 /* Enable PCK output */
168 pmc_disable_pck(PMC_PCK_3);
169 pmc_switch_pck_to_mck(PMC_PCK_3, PMC_PCK_PRES(0));
170 pmc_enable_pck(PMC_PCK_3);
171#endif
172
173 /* Init TC to capture mode. */
174 tc_init(TC, TC_CHANNEL_CAPTURE,
175 TC_CAPTURE_TIMER_SELECTION /* Clock Selection */
176 | TC_CMR_LDRA_RISING /* RA Loading: rising edge of TIOA */
177 | TC_CMR_LDRB_FALLING /* RB Loading: falling edge of TIOA */
178 | TC_CMR_ABETRG /* External Trigger: TIOA */
179 | TC_CMR_ETRGEDG_FALLING /* External Trigger Edge: Falling edge */
180 );
181}

188void TC_Handler(void)
189{
192 if ((tc_get_status(TC, TC_CHANNEL_CAPTURE) & TC_SR_LDRBS) == TC_SR_LDRBS) {
194 gs_ul_captured_pulses++;
196 gs_ul_captured_ra = tc_read_ra(TC, TC_CHANNEL_CAPTURE);
199 gs_ul_captured_rb = tc_read_rb(TC, TC_CHANNEL_CAPTURE);

231int main(void)
232{
233 uint8_t key;
234 uint16_t frequence, dutycycle;
235
236 /* Initialize the SAM system */
237 sysclk_init();
238 board_init();
239
240 /* Initialize the console uart */
241 configure_console();
242
243 /* Output example information */
244 printf("-- TC capture waveform Example --\r\n");
245 printf("-- %s\n\r", BOARD_NAME);
246 printf("-- Compiled: %s %s --\n\r", __DATE__, __TIME__);
247
249
250 ioport_set_pin_mode(PIN_TC_WAVEFORM, PIN_TC_WAVEFORM_MUX);
252 ioport_disable_pin(PIN_TC_WAVEFORM);
254
256
257 ioport_set_pin_mode(PIN_TC_CAPTURE, PIN_TC_CAPTURE_MUX);
259 ioport_disable_pin(PIN_TC_CAPTURE);
261
262 /* Configure TC TC_CHANNEL_WAVEFORM as waveform operating mode */
263 printf("Configure TC%d channel %d as waveform operating mode \n\r",
264 TC_PERIPHERAL, TC_CHANNEL_WAVEFORM);
266 tc_waveform_initialize();
268
269 /* Configure TC TC_CHANNEL_CAPTURE as capture operating mode */
270 printf("Configure TC%d channel %d as capture operating mode \n\r",
271 TC_PERIPHERAL, TC_CHANNEL_CAPTURE);
273 tc_capture_initialize();
275
277
278 NVIC_DisableIRQ(TC_IRQn);
279 NVIC_ClearPendingIRQ(TC_IRQn);
280 NVIC_SetPriority(TC_IRQn, 0);
281 NVIC_EnableIRQ(TC_IRQn);
283
284 /* Display menu */
285 display_menu();
286
287 while (1) {
288 scanf("%c", (char *)&key);
289
290 switch (key) {
291 case 'h':
292 display_menu();
293 break;
294
295 case 's':
296 if (gs_ul_captured_pulses) {
297 tc_disable_interrupt(TC, TC_CHANNEL_CAPTURE, TC_IDR_LDRBS);
298 printf("Captured %u pulses from TC%d channel %d, RA = %u, RB = %u \n\r",
299 (unsigned)gs_ul_captured_pulses, TC_PERIPHERAL,
300 TC_CHANNEL_CAPTURE, (unsigned)gs_ul_captured_ra,
301 (unsigned)gs_ul_captured_rb);
302 frequence = (sysclk_get_peripheral_bus_hz(TC) /
303 divisors[TC_CAPTURE_TIMER_SELECTION]) /
304 gs_ul_captured_rb;
305 dutycycle
306 = (gs_ul_captured_rb - gs_ul_captured_ra) * 100 /
307 gs_ul_captured_rb;
308 printf("Captured wave frequency = %d Hz, Duty cycle = %d%% \n\r",
309 frequence, dutycycle);
310
311 gs_ul_captured_pulses = 0;
312 gs_ul_captured_ra = 0;
313 gs_ul_captured_rb = 0;
314 } else {
315 puts("No waveform has been captured\r");
316 }
317
318 puts("\n\rPress 'h' to display menu\r");
319 break;
320
321 case 'c':
322 puts("Start capture, press 's' to stop \r");
324 tc_enable_interrupt(TC, TC_CHANNEL_CAPTURE, TC_IER_LDRBS);
326 /* Start the timer counter on TC TC_CHANNEL_CAPTURE */
328 tc_start(TC, TC_CHANNEL_CAPTURE);
330 break;
331
332 default:
333 /* Set waveform configuration #n */
334 if ((key >= '0') && (key <= ('0' + gc_uc_nbconfig - 1))) {
335 if (!gs_ul_captured_pulses) {
336 gs_uc_configuration = key - '0';
337 tc_waveform_initialize();
338 } else {
339 puts("Capturing ... , press 's' to stop capture first \r");
340 }
341 }
342
343 break;
344 }
345 }
346}
