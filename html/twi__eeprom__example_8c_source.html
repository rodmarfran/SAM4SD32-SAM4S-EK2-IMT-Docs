210int main(void)
211{
212 uint32_t i;
213 twi_options_t opt;
214 twi_packet_t packet_tx, packet_rx;
215
216 /* Initialize the SAM system */
217 sysclk_init();
218
219 /* Initialize the board */
220 board_init();
221
222 /* Turn off LEDs */
223#if SAM3XA
224 LED_Off(LED0_GPIO);
225 LED_Off(LED1_GPIO);
226#else
227 LED_Off(LED0);
228#endif
229 /* Initialize the console UART */
230 configure_console();
231
232 /* Output example information */
233 puts(STRING_HEADER);
234
235 /* Configure systick for 1 ms */
236 puts("Configure system tick to get 1ms tick period.\r");
237 if (SysTick_Config(sysclk_get_cpu_hz() / 1000)) {
238 puts("-E- Systick configuration error\r");
239 while (1) {
240 /* Capture error */
241 }
242 }
243
244#if (SAMG55)
245 /* Enable the peripheral and set TWI mode. */
246 flexcom_enable(BOARD_FLEXCOM_TWI);
247 flexcom_set_opmode(BOARD_FLEXCOM_TWI, FLEXCOM_TWI);
248#else
249 /* Enable the peripheral clock for TWI */
250 pmc_enable_periph_clk(BOARD_ID_TWI_EEPROM);
251#endif
252
253 /* Configure the options of TWI driver */
254 opt.master_clk = sysclk_get_peripheral_hz();
255 opt.speed = TWI_CLK;
256
257 /* Configure the data packet to be transmitted */
258 packet_tx.chip = AT24C_ADDRESS;
259 packet_tx.addr[0] = EEPROM_MEM_ADDR >> 8;
260 packet_tx.addr[1] = EEPROM_MEM_ADDR;
261 packet_tx.addr_length = EEPROM_MEM_ADDR_LENGTH;
262 packet_tx.buffer = (uint8_t *) test_data_tx;
263 packet_tx.length = TEST_DATA_LENGTH;
264
265 /* Configure the data packet to be received */
266 packet_rx.chip = packet_tx.chip;
267 packet_rx.addr[0] = packet_tx.addr[0];
268 packet_rx.addr[1] = packet_tx.addr[1];
269 packet_rx.addr_length = packet_tx.addr_length;
270 packet_rx.buffer = gs_uc_test_data_rx;
271 packet_rx.length = packet_tx.length;
272
273 if (twi_master_init(BOARD_BASE_TWI_EEPROM, &opt) != TWI_SUCCESS) {
274 puts("-E-\tTWI master initialization failed.\r");
275#if SAM3XA
276 LED_On(LED0_GPIO);
277 LED_On(LED1_GPIO);
278#endif
279 while (1) {
280 /* Capture error */
281 }
282 }
283
284 /* Send test pattern to EEPROM */
285 if (twi_master_write(BOARD_BASE_TWI_EEPROM, &packet_tx) != TWI_SUCCESS) {
286 puts("-E-\tTWI master write packet failed.\r");
287#if SAM3XA
288 LED_On(LED0_GPIO);
289 LED_On(LED1_GPIO);
290#endif
291 while (1) {
292 /* Capture error */
293 }
294 }
295 printf("Write:\tOK!\n\r");
296
297 /* Wait at least 10 ms */
298 mdelay(WAIT_TIME);
299
300 /* Get memory from EEPROM */
301 if (twi_master_read(BOARD_BASE_TWI_EEPROM, &packet_rx) != TWI_SUCCESS) {
302 puts("-E-\tTWI master read packet failed.\r");
303#if SAM3XA
304 LED_On(LED0_GPIO);
305 LED_On(LED1_GPIO);
306#endif
307 while (1) {
308 /* Capture error */
309 }
310 }
311 puts("Read:\tOK!\r");
312
313 /* Compare the sent and the received */
314 for (i = 0; i < TEST_DATA_LENGTH; i++) {
315 if (test_data_tx[i] != gs_uc_test_data_rx[i]) {
316 /* No match */
317 puts("Data comparison:\tUnmatched!\r");
318#if SAM3XA
319 LED_On(LED0_GPIO);
320#endif
321 while (1) {
322 /* Capture error */
323 }
324 }
325 }
326 /* Match */
327 puts("Data comparison:\tMatched!\r");
328#if SAM3XA
329 LED_On(LED1_GPIO);
330#else
331 LED_On(LED0);
332#endif
333 while (1) {
334 }
335}
