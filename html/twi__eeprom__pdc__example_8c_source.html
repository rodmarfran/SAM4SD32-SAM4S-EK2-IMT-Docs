231static void twi_master_pdc_write(Twi *p_twi, twi_packet_t *p_packet)
232{
233 pdc_packet_t pdc_twi_packet;
234 uint8_t *buffer = p_packet->buffer;
235
236 pdc_disable_transfer(g_p_twim_pdc, PERIPH_PTCR_TXTDIS | PERIPH_PTCR_RXTDIS);
237
238 pdc_twi_packet.ul_addr = (uint32_t)buffer;
239#if SAMG53
240 pdc_twi_packet.ul_size = (p_packet->length - 1);
241#else
242 pdc_twi_packet.ul_size = (p_packet->length);
243#endif
244 pdc_tx_init(g_p_twim_pdc, &pdc_twi_packet, NULL);
245
246 /* Set write mode, slave address and 3 internal address byte lengths */
247 p_twi->TWI_MMR = 0;
248 p_twi->TWI_MMR = TWI_MMR_DADR(p_packet->chip) |
249 ((p_packet->addr_length << TWI_MMR_IADRSZ_Pos) &
250 TWI_MMR_IADRSZ_Msk);
251
252 /* Set internal address for remote chip */
253 p_twi->TWI_IADR = 0;
254 p_twi->TWI_IADR = twi_mk_addr(p_packet->addr, p_packet->addr_length);
255
256 Write_Interrupt_Flag = 1;
257
258 twi_enable_interrupt(BOARD_BASE_TWI_EEPROM, TWI_SR_TXCOMP);
259
260 /* Enable the TX PDC transfer requests */
261 pdc_enable_transfer(g_p_twim_pdc, PERIPH_PTCR_TXTEN);
262
263 /* Waiting transfer done*/
264 while((twi_read_status(BOARD_BASE_TWI_EEPROM) & TWI_SR_ENDTX) == 0);
265
266 /* Disable the RX and TX PDC transfer requests */
267 pdc_disable_transfer(g_p_twim_pdc, PERIPH_PTCR_TXTDIS | PERIPH_PTCR_RXTDIS);
268
269 while((twi_read_status(BOARD_BASE_TWI_EEPROM) & TWI_SR_TXRDY) == 0);
270
271 p_twi->TWI_CR = TWI_CR_STOP;
272#if SAMG53
273 p_twi->TWI_THR = *(buffer + p_packet->length - 1);
274#endif
275 while (!(p_twi->TWI_SR & TWI_SR_TXCOMP)) {
276 }
277}

285static void twi_master_pdc_read(Twi *p_twi, twi_packet_t *p_packet)
286{
287 pdc_packet_t pdc_twi_packet;
288 uint8_t *buffer = p_packet->buffer;
289
290 pdc_disable_transfer(g_p_twim_pdc, PERIPH_PTCR_TXTDIS | PERIPH_PTCR_RXTDIS);
291
292 pdc_twi_packet.ul_addr = (uint32_t)buffer;
293#if SAMG53
294 pdc_twi_packet.ul_size = (p_packet->length - 2);
295#else
296 pdc_twi_packet.ul_size = (p_packet->length);
297#endif
298 pdc_rx_init(g_p_twim_pdc, &pdc_twi_packet, NULL);
299
300 /* Set write mode, slave address and 2 internal address byte lengths */
301 p_twi->TWI_MMR = 0;
302 p_twi->TWI_MMR = TWI_MMR_MREAD | TWI_MMR_DADR(p_packet->chip) |
303 ((p_packet->addr_length << TWI_MMR_IADRSZ_Pos) &
304 TWI_MMR_IADRSZ_Msk);
305
306 /* Set internal address for remote chip */
307 p_twi->TWI_IADR = 0;
308 p_twi->TWI_IADR = twi_mk_addr(p_packet->addr, p_packet->addr_length);
309
310 Read_Interrupt_Flag = 1;
311
312 twi_enable_interrupt(BOARD_BASE_TWI_EEPROM, TWI_SR_TXCOMP);
313
314 /* Enable the RX PDC transfer requests */
315 pdc_enable_transfer(g_p_twim_pdc, PERIPH_PTCR_RXTEN);
316
317 p_twi->TWI_CR = TWI_CR_START;
318
319 /* Waiting transfer done*/
320 while((twi_read_status(BOARD_BASE_TWI_EEPROM) & TWI_SR_ENDRX) == 0);
321
322 /* Disable the RX PDC transfer requests */
323 pdc_disable_transfer(g_p_twim_pdc, PERIPH_PTCR_RXTDIS);
324
325 while((twi_read_status(BOARD_BASE_TWI_EEPROM) & TWI_SR_RXRDY) == 0);
326
327 p_twi->TWI_CR = TWI_CR_STOP;
328#if SAMG53
329 *(buffer + p_packet->length - 2) = p_twi->TWI_RHR;
330
331 while((twi_read_status(BOARD_BASE_TWI_EEPROM) & TWI_SR_RXRDY) == 0);
332
333 *(buffer + p_packet->length - 1) = p_twi->TWI_RHR;
334#endif
335 while (!(p_twi->TWI_SR & TWI_SR_TXCOMP)) {
336 }

339void BOARD_TWI_Handler(void)
340{
341 uint32_t status;
342
343 status = twi_get_interrupt_mask(BOARD_BASE_TWI_EEPROM);
344
345 if(((status & TWI_IMR_TXCOMP) == TWI_IMR_TXCOMP)
346 && (Write_Interrupt_Flag == 1)) {
347 printf("PDC Write transfer:\tOK!\n\r");
348 Write_Interrupt_Flag = 0;
349 twi_disable_interrupt(BOARD_BASE_TWI_EEPROM, TWI_SR_TXCOMP);
350 } else if(((status & TWI_IMR_TXCOMP) == TWI_IMR_TXCOMP)
351 &&(Read_Interrupt_Flag == 1)) {
352 printf("PDC Read transfer:\tOK!\n\r");
353 Read_Interrupt_Flag = 0;
354 twi_disable_interrupt(BOARD_BASE_TWI_EEPROM, TWI_IMR_TXCOMP);
355 }

363int main(void)
364{
365 uint32_t i;

368 /* Initialize the SAM system */
369 sysclk_init();
370
371 /* Initialize the board */
372 board_init();
373
374 /* Turn off LEDs */
375#if SAM3XA
376 LED_Off(LED0_GPIO);
377 LED_Off(LED1_GPIO);
378#else
379 LED_Off(LED0);
380#endif
381 /* Initialize the console UART */
382 configure_console();
383
384 /* Output example information */
385 puts(STRING_HEADER);
386
387 /* Configure systick for 1 ms */
388 puts("Configure system tick to get 1ms tick period.\r");
389 if (SysTick_Config(sysclk_get_cpu_hz() / 1000)) {
390 puts("-E- Systick configuration error\r");
391 while (1) {
392 /* Capture error */
393 }
394 }
395
396#if SAMG55
397 /* Enable the peripheral and set TWI mode. */
398 flexcom_enable(BOARD_FLEXCOM_TWI);
399 flexcom_set_opmode(BOARD_FLEXCOM_TWI, FLEXCOM_TWI);
400#else
401 /* Enable the peripheral clock for TWI */
402 pmc_enable_periph_clk(BOARD_ID_TWI_EEPROM);
403#endif
404
405 /* Configure the options of TWI driver */
406 opt.master_clk = sysclk_get_peripheral_hz();
407 opt.speed = TWI_CLK;
408
409 /* Configure the data packet to be transmitted */
410 packet_tx.chip = AT24C_ADDRESS;
411 packet_tx.addr[0] = EEPROM_MEM_ADDR >> 8;
412 packet_tx.addr[1] = EEPROM_MEM_ADDR;
413 packet_tx.addr_length = EEPROM_MEM_ADDR_LENGTH;
414 packet_tx.buffer = (uint8_t *) test_data_tx;
415 packet_tx.length = TEST_DATA_LENGTH;
416
417 /* Configure the data packet to be received */
418 packet_rx.chip = packet_tx.chip;
419 packet_rx.addr[0] = packet_tx.addr[0];
420 packet_rx.addr[1] = packet_tx.addr[1];
421 packet_rx.addr_length = packet_tx.addr_length;
422 packet_rx.buffer = gs_uc_test_data_rx;
423 packet_rx.length = packet_tx.length;
424
425 if (twi_master_init(BOARD_BASE_TWI_EEPROM, &opt) != TWI_SUCCESS) {
426 puts("-E-\tTWI master initialization failed.\r");
427#if SAM3XA
428 LED_On(LED0_GPIO);
429 LED_On(LED1_GPIO);
430#endif
431 while (1) {
432 /* Capture error */
433 }
434 }
435
436 /* Configure TWI interrupts */
437 NVIC_DisableIRQ(BOARD_TWI_IRQn);
438 NVIC_ClearPendingIRQ(BOARD_TWI_IRQn);
439 NVIC_SetPriority(BOARD_TWI_IRQn, 0);
440 NVIC_EnableIRQ(BOARD_TWI_IRQn);
441
442 /* Get pointer to TWI master PDC register base */
443 g_p_twim_pdc = twi_get_pdc_base(BOARD_BASE_TWI_EEPROM);
444
445#if SAMG55
446 twi_enable_alternative_command(BOARD_BASE_TWI_EEPROM);
447#endif
448 twi_master_pdc_write(BOARD_BASE_TWI_EEPROM, &packet_tx);
449
450 /* Wait at least 10 ms */
451 mdelay(WAIT_TIME);
452
453 twi_master_pdc_read(BOARD_BASE_TWI_EEPROM, &packet_rx);
454
455 /* Compare the sent and the received */
456 for (i = 0; i < (TEST_DATA_LENGTH); i++) {
457 if (test_data_tx[i] != gs_uc_test_data_rx[i]) {
458 /* No match */
459 puts("Data comparison:\tUnmatched!\r");
460#if SAM3XA
461 LED_On(LED0_GPIO);
462#endif
463 while (1) {
464 /* Capture error */
465 }
466 }
467 }
468 /* Match */
469 puts("Data comparison:\tMatched!\r");
470#if SAM3XA
471 LED_On(LED1_GPIO);
472#else
473 LED_On(LED0);
474#endif
475 while (1) {
476 }
477}
