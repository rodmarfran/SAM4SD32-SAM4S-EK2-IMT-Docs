166 twi_package_t packet = {
171#if defined(sam3xek) || defined(arduinoduex) || defined(sam4cek) || defined(sam4cmpdb) || defined(sam4cmsdb)
172 .addr[0] = EEPROM_MEM_ADDR >> 8, // TWI slave memory address data MSB
173 .addr[1] = EEPROM_MEM_ADDR, // TWI slave memory address data LSB
174 .addr_length = sizeof (uint16_t), // TWI slave memory address data size

184 // Perform a multi-byte write access then check the result.
185 while (twi_master_write(TWI_EXAMPLE, &packet) != TWI_SUCCESS);

187 uint8_t data_received[PATTERN_TEST_LENGTH] = {0};

189 twi_package_t packet_received = {
190#if defined(sam3xek) || defined(arduinoduex) || defined(sam4cek) || defined(sam4cmpdb) || defined(sam4cmsdb)
191 .addr[0] = EEPROM_MEM_ADDR >> 8, // TWI slave memory address data MSB
192 .addr[1] = EEPROM_MEM_ADDR, // TWI slave memory address data LSB
193 .addr_length = sizeof (uint16_t), // TWI slave memory address data size
194#else
195 .addr[0] = EEPROM_MEM_ADDR, // TWI slave memory address data MSB
196 .addr_length = sizeof (uint8_t), // TWI slave memory address data size
197#endif
198 .chip = EEPROM_BUS_ADDR, // TWI slave bus address
199 .buffer = data_received, // transfer data destination buffer
200 .length = PATTERN_TEST_LENGTH // transfer data size (bytes)

203 // Perform a multi-byte read access then check the result.

206 // Verify that the received data matches the sent data.
207 for (uint32_t i = 0 ; i < PATTERN_TEST_LENGTH; ++i) {

215 //test PASS
