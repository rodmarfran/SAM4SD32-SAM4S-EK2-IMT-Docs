124void BOARD_TWI_Handler(void)
125{
126 uint32_t status;
127
128 status = twi_get_interrupt_status(BOARD_BASE_TWI_SLAVE);
129
130 if (((status & TWI_SR_SVACC) == TWI_SR_SVACC)
131 && (emulate_driver.uc_acquire_address == 0)) {
132 twi_disable_interrupt(BOARD_BASE_TWI_SLAVE, TWI_IDR_SVACC);
133 twi_enable_interrupt(BOARD_BASE_TWI_SLAVE, TWI_IER_RXRDY | TWI_IER_GACC
134 | TWI_IER_NACK | TWI_IER_EOSACC | TWI_IER_SCL_WS);
135 emulate_driver.uc_acquire_address++;
136 emulate_driver.us_page_address = 0;
137 emulate_driver.us_offset_memory = 0;
138 }
139
140 if ((status & TWI_SR_GACC) == TWI_SR_GACC) {
141 puts("General Call Treatment\n\r");
142 puts("not treated");
143 }
144
145 if (((status & TWI_SR_SVACC) == TWI_SR_SVACC) && ((status & TWI_SR_GACC) == 0)
146 && ((status & TWI_SR_RXRDY) == TWI_SR_RXRDY)) {
147
148 if (emulate_driver.uc_acquire_address == 1) {
149 /* Acquire MSB address */
150 emulate_driver.us_page_address =
151 (twi_read_byte(BOARD_BASE_TWI_SLAVE) & 0xFF) << 8;
152 emulate_driver.uc_acquire_address++;
153 } else {
154 if (emulate_driver.uc_acquire_address == 2) {
155 /* Acquire LSB address */
156 emulate_driver.us_page_address |=
157 (twi_read_byte(BOARD_BASE_TWI_SLAVE) & 0xFF);
158 emulate_driver.uc_acquire_address++;
159 } else {
160 /* Read one byte of data from master to slave device */
161 emulate_driver.uc_memory[emulate_driver.us_page_address +
162 emulate_driver.us_offset_memory] =
163 (twi_read_byte(BOARD_BASE_TWI_SLAVE) & 0xFF);
164
165 emulate_driver.us_offset_memory++;
166 }
167 }
168 } else {
169 if (((status & TWI_SR_TXRDY) == TWI_SR_TXRDY)
170 && ((status & TWI_SR_TXCOMP) == TWI_SR_TXCOMP)
171 && ((status & TWI_SR_EOSACC) == TWI_SR_EOSACC)) {
172 /* End of transfer, end of slave access */
173 emulate_driver.us_offset_memory = 0;
174 emulate_driver.uc_acquire_address = 0;
175 emulate_driver.us_page_address = 0;
176 twi_enable_interrupt(BOARD_BASE_TWI_SLAVE, TWI_SR_SVACC);
177 twi_disable_interrupt(BOARD_BASE_TWI_SLAVE,
178 TWI_IDR_RXRDY | TWI_IDR_GACC |
179 TWI_IDR_NACK | TWI_IDR_EOSACC | TWI_IDR_SCL_WS);
180 } else {
181 if (((status & TWI_SR_SVACC) == TWI_SR_SVACC)
182 && ((status & TWI_SR_GACC) == 0)
183 && (emulate_driver.uc_acquire_address == 3)
184 && ((status & TWI_SR_SVREAD) == TWI_SR_SVREAD)
185 && ((status & TWI_SR_TXRDY) == TWI_SR_TXRDY)
186 && ((status & TWI_SR_NACK) == 0)) {
187 /* Write one byte of data from slave to master device */
188 twi_write_byte(BOARD_BASE_TWI_SLAVE,
189 emulate_driver.uc_memory[emulate_driver.us_page_address
190 + emulate_driver.us_offset_memory]);
191 emulate_driver.us_offset_memory++;
192 }
193 }
194 }
195}

223int main(void)
224{
225 uint32_t i;
226
227 /* Initialize the SAM system */
228 sysclk_init();
229
230#if (SAM4S || SAM4E)
231 /* Select PB4 and PB5 function, this will cause JTAG discconnect */
232 REG_CCFG_SYSIO |= CCFG_SYSIO_SYSIO4;
233 REG_CCFG_SYSIO |= CCFG_SYSIO_SYSIO5;
234#endif
235
236 /* Initialize the board */
237 board_init();
238
239 /* Initialize the console UART */
240 configure_console();
241
242 /* Output example information */
243 puts(STRING_HEADER);
244
245#if (SAMG55)
246 /* Enable the peripheral and set TWI mode. */
247 flexcom_enable(BOARD_FLEXCOM_TWI);
248 flexcom_set_opmode(BOARD_FLEXCOM_TWI, FLEXCOM_TWI);
249#else
250 /* Enable the peripheral clock for TWI */
251 pmc_enable_periph_clk(BOARD_ID_TWI_SLAVE);
252#endif
253
254 for (i = 0; i < MEMORY_SIZE; i++) {
255 emulate_driver.uc_memory[i] = 0;
256 }
257 emulate_driver.us_offset_memory = 0;
258 emulate_driver.uc_acquire_address = 0;
259 emulate_driver.us_page_address = 0;
260
261 /* Configure TWI as slave */
262 puts("-I- Configuring the TWI in slave mode\n\r");
263 twi_slave_init(BOARD_BASE_TWI_SLAVE, SLAVE_ADDRESS);
264
265 /* Clear receipt buffer */
266 twi_read_byte(BOARD_BASE_TWI_SLAVE);
267
268 /* Configure TWI interrupts */
269 NVIC_DisableIRQ(BOARD_TWI_IRQn);
270 NVIC_ClearPendingIRQ(BOARD_TWI_IRQn);
271 NVIC_SetPriority(BOARD_TWI_IRQn, 0);
272 NVIC_EnableIRQ(BOARD_TWI_IRQn);
273 twi_enable_interrupt(BOARD_BASE_TWI_SLAVE, TWI_SR_SVACC);
274
275 while (1) {
276 }
277}
