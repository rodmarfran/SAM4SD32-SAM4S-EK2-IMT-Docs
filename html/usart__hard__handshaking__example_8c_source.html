155void USART_Handler(void)
156{
157 uint32_t ul_status;
158
159 tc_stop(TC0, 0);
160
161 /* Read USART status. */
162 ul_status = usart_get_status(BOARD_USART);
163
164 /* Receive buffer is full. */
165 if (ul_status & US_CSR_RXBUFF) {
166 g_ul_bytes_received += 2 * BUFFER_SIZE;
167 if (g_ul_bytes_received < MAX_BPS) {
168 /* Restart transfer if BPS is not high enough. */
169 g_st_packet.ul_addr = (uint32_t)gs_puc_buffer;
170 g_st_packet.ul_size = BUFFER_SIZE;
171 g_st_nextpacket.ul_addr = (uint32_t)gs_puc_nextbuffer;
172 g_st_nextpacket.ul_size = BUFFER_SIZE;
173 pdc_rx_init(g_p_pdc, &g_st_packet, &g_st_nextpacket);
174 } else {
175 /* Otherwise disable interrupt. */
176 usart_disable_interrupt(BOARD_USART, US_IDR_RXBUFF);
177 }
178 memcpy(gs_dump_buffer, gs_puc_buffer, BUFFER_SIZE);
179 }
180 tc_start(TC0, 0);
181}

190void TC0_Handler(void)
191{
192 uint32_t ul_status;
193 static uint32_t bytes_total = 0;
194
195 /* Read TC0 status. */
196 ul_status = tc_get_status(TC0, 0);
197
198 /* RC compare. */
199 if ((ul_status & TC_SR_CPCS) == TC_SR_CPCS) {
200 if((pdc_read_rx_counter(g_p_pdc) != 0) &&
201 (pdc_read_rx_counter(g_p_pdc) != BUFFER_SIZE)) {
202 if(pdc_read_rx_next_counter(g_p_pdc) == 0) {
203 g_ul_bytes_received += (2 * BUFFER_SIZE - pdc_read_rx_counter(g_p_pdc));
204 } else {
205 g_ul_bytes_received += (BUFFER_SIZE - pdc_read_rx_counter(g_p_pdc));
206 }
207 memset(gs_dump_buffer, 0, BUFFER_SIZE);
208 memcpy(gs_dump_buffer, gs_puc_buffer, BUFFER_SIZE -
209 pdc_read_rx_counter(g_p_pdc));
210 } else if((pdc_read_rx_counter(g_p_pdc) == BUFFER_SIZE) &&
211 (pdc_read_rx_next_counter(g_p_pdc) == 0)) {
212 g_ul_bytes_received += BUFFER_SIZE;
213 memcpy(gs_dump_buffer, gs_puc_buffer, BUFFER_SIZE);
214 }
215
216 /* Display info. */
217 bytes_total += g_ul_bytes_received;
218 memset(g_puc_string, 0, BUFFER_SIZE);
219 sprintf((char *)g_puc_string, "Bps: %4lu; Tot: %6lu\r\n",
220 (unsigned long)g_ul_bytes_received,
221 (unsigned long)bytes_total);
222 usart_write_line(BOARD_USART, (char *)g_puc_string);
223 /* Resume transfer */
224 g_st_packet.ul_addr = (uint32_t)gs_puc_buffer;
225 g_st_packet.ul_size = BUFFER_SIZE;
226 g_st_nextpacket.ul_addr = (uint32_t)gs_puc_nextbuffer;
227 g_st_nextpacket.ul_size = BUFFER_SIZE;
228 pdc_rx_init(g_p_pdc, &g_st_packet, &g_st_nextpacket);
229 usart_enable_interrupt(BOARD_USART, US_IER_RXBUFF);
230
231 g_ul_bytes_received = 0;
232 }

238static void configure_tc(void)
239{
240 uint32_t ul_div;
241 uint32_t ul_tcclks;
242 uint32_t ul_sysclk;
243
244 /* Get system clock. */
245 ul_sysclk = sysclk_get_cpu_hz();
246
247 /* Configure PMC. */
248 pmc_enable_periph_clk(ID_TC0);
249
251 tc_find_mck_divisor(TC_FREQ, ul_sysclk, &ul_div, &ul_tcclks, ul_sysclk);
252 tc_init(TC0, 0, ul_tcclks | TC_CMR_CPCTRG);
253 tc_write_rc(TC0, 0, (ul_sysclk / ul_div) / TC_FREQ);
254
255 /* Configure and enable interrupt on RC compare. */
256 NVIC_EnableIRQ((IRQn_Type) ID_TC0);
257 tc_enable_interrupt(TC0, 0, TC_IER_CPCS);
258}

263static void configure_usart(void)
264{
265 static uint32_t ul_sysclk;
266 const sam_usart_opt_t usart_console_settings = {
267 BOARD_USART_BAUDRATE,
268 US_MR_CHRL_8_BIT,
269 US_MR_PAR_NO,
270 US_MR_NBSTOP_1_BIT,
271 US_MR_CHMODE_NORMAL,
272 /* This field is only used in IrDA mode. */
273 0
274 };
275
276 /* Get peripheral clock. */
277 ul_sysclk = sysclk_get_peripheral_hz();
278
279 /* Enable peripheral clock. */
280 sysclk_enable_peripheral_clock(BOARD_ID_USART);
281
282 /* Configure USART. */
283 usart_init_hw_handshaking(BOARD_USART, &usart_console_settings, ul_sysclk);
284
285 /* Disable all the interrupts. */
286 usart_disable_interrupt(BOARD_USART, ALL_INTERRUPT_MASK);
287
288 /* Enable TX & RX function. */
289 usart_enable_tx(BOARD_USART);
290 usart_enable_rx(BOARD_USART);
291
292 /* Specify that stdout should not be buffered. */
293#if defined(__GNUC__)
294 setbuf(stdout, NULL);
295#else
296 /* Already the case in IAR's Normal DLIB default configuration: printf()
297 * emits one character at a time.
298 */
299#endif
300
301 /* Configure and enable interrupt of USART. */
302 NVIC_EnableIRQ(USART_IRQn);
303}

325int main(void)
326{
327 uint8_t uc_char;
328 uint8_t uc_flag;
329
330 /* Initialize the system. */
331 sysclk_init();
332 board_init();
333
334 /* Configure UART for debug message output. */
335 configure_console();
336
337 /* Output example information. */
338 puts(STRING_HEADER);
339
340 /* Initialize board USART. */
341 configure_usart();
342
343 /* Initialize TC0. */
344 configure_tc();
345
346 /* Get board USART PDC base address and enable receiver. */
347 g_p_pdc = usart_get_pdc_base(BOARD_USART);
348 g_st_packet.ul_addr = (uint32_t)gs_puc_buffer;
349 g_st_packet.ul_size = BUFFER_SIZE;
350 g_st_nextpacket.ul_addr = (uint32_t)gs_puc_nextbuffer;
351 g_st_nextpacket.ul_size = BUFFER_SIZE;
352 pdc_rx_init(g_p_pdc, &g_st_packet, &g_st_nextpacket);
353 pdc_enable_transfer(g_p_pdc, PERIPH_PTCR_RXTEN);
354
355 usart_enable_interrupt(BOARD_USART, US_IER_RXBUFF);
356
357 /* Start the Timer counter. */
358 tc_start(TC0, 0);
359
360 while (1) {
361 uc_char = 0;
362 uc_flag = uart_read(CONSOLE_UART, &uc_char);
363 if (!uc_flag) {
364 switch (uc_char) {
365 case 'd':
366 case 'D':
367 usart_write_line(BOARD_USART, (char *)gs_dump_buffer);
368 break;
369 default:
370 break;
371 }
372 }
373 }
