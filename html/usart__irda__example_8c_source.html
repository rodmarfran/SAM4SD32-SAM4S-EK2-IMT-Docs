140void USART_Handler(void)
141{
142 uint32_t ul_status;
143
144 /* Read USART Status. */
145 ul_status = usart_get_status(BOARD_USART);
146
147 /* Receiving is done. */
148 if ((ul_status & US_CSR_RXBUFF) && (g_uc_state == STATE_RECEIVE)) {
149 g_ul_recv_done = true;
150 usart_disable_interrupt(BOARD_USART, US_IDR_RXBUFF);
151 }
152
153 /* Transmitting is done. */
154 if ((ul_status & US_CSR_TXBUFE) && (g_uc_state == STATE_TRANSMIT)) {
155 g_ul_sent_done = true;
156 usart_disable_interrupt(BOARD_USART, US_IDR_TXBUFE);
157 }
158}

178static void func_receiver(void)
179{
180 uint32_t ul_temp;
181
182 /* Disable Transmitter. */
183 usart_disable_tx(BOARD_USART);
184
185 /* Configure the TXD pin as PIO. */
186 pio_configure_pin(PIN_USART_TXD_IDX, PIN_USART_TXD_IO_FLAGS);
187 pio_set_pin_low(PIN_USART_TXD_IDX);
188
189 /* Enable receiver. */
190 usart_enable_rx(BOARD_USART);
191
192 /* Read dummy to make sure that there are no characters in US_THR! */
193 if (usart_is_rx_ready(BOARD_USART)) {
194 usart_read(BOARD_USART, &ul_temp);
195 /* avoid Cppcheck Warning */
196 UNUSED(ul_temp);
197 }
198}

218static void configure_usart(void)
219{
220 const sam_usart_opt_t usart_console_settings = {
221 BOARD_USART_BAUDRATE,
222 US_MR_CHRL_8_BIT,
223 US_MR_PAR_NO,
224 US_MR_NBSTOP_1_BIT,
225 US_MR_CHMODE_NORMAL,
226 BOARD_USART_IRDA_FILTER
227 };
228
229 /* Enable peripheral clock. */
230 sysclk_enable_peripheral_clock(BOARD_ID_USART);
231
232 /* Configure USART in IrDA mode. */
233 usart_init_irda(BOARD_USART, &usart_console_settings, sysclk_get_peripheral_hz());
234
235 /* Disable all the interrupts. */
236 usart_disable_interrupt(BOARD_USART, ALL_INTERRUPT_MASK);
237
238 /* Enable TX & RX function. */
239 usart_enable_tx(BOARD_USART);
240 usart_enable_rx(BOARD_USART);
241
242 /* Configure and enable interrupt of USART. */
243 NVIC_EnableIRQ(USART_IRQn);
244}

268int main(void)
269{
270 uint8_t uc_char;
271
272 /* Initialize the SAM system. */
273 sysclk_init();
274 board_init();
275
276 /* Configure UART for debug message output. */
277 configure_console();
278
279 /* Output example information. */
280 puts(STRING_HEADER);
281
282 /* Initialize board USART. */
283 configure_usart();
284
285 /* Get board USART PDC base address and enable receiver and transmitter. */
286 g_p_pdc = usart_get_pdc_base(BOARD_USART);
287 pdc_enable_transfer(g_p_pdc, PERIPH_PTCR_RXTEN | PERIPH_PTCR_TXTEN);
288
289 puts("-I- Press t to transmit, press r to receive...\r");
290
291 /* Main loop. */
292 while (1) {
293 uc_char = 0;
294 uart_read(CONSOLE_UART, (uint8_t *)&uc_char);
295
296 switch (uc_char) {
297 case 't':
298 case 'T':
299 g_uc_state = STATE_TRANSMIT;
300 /* Enable transmitter. */
301 func_transmitter();
302 g_st_packet.ul_addr = (uint32_t)g_uc_send_data;
303 g_st_packet.ul_size = BUFFER_SIZE;
304 pdc_tx_init(g_p_pdc, &g_st_packet, NULL);
305 usart_enable_interrupt(BOARD_USART, US_IER_TXBUFE);
306 while (!g_ul_sent_done) {
307 }
308
309 puts("-I- Sent Done!\r");
310 g_ul_sent_done = false;
311 g_uc_state = STATE_IDLE;
312 puts("-I- Press t to transmit, press r to receive...\r");
313 break;
314
315 case 'r':
316 case 'R':
317 g_uc_state = STATE_RECEIVE;
318
319 /* Enable receiver. */
320 puts("-I- IrDA receive mode\r");
321 func_receiver();
322 g_st_packet.ul_addr = (uint32_t)g_uc_recv_data;
323 g_st_packet.ul_size = BUFFER_SIZE;
324 pdc_rx_init(g_p_pdc, &g_st_packet, NULL);
325 usart_enable_interrupt(BOARD_USART, US_IER_RXBUFF);
326
327 while (!g_ul_recv_done) {
328 }
329
330 puts("-I- Received Done! \r");
331 dump_recv_buf(g_uc_recv_data, BUFFER_SIZE);
332 memset(g_uc_recv_data, 0, BUFFER_SIZE);
333 g_uc_state = STATE_IDLE;
334 g_ul_recv_done = false;
335 puts("-I- Press t to transmit, press r to receive...\r");
336 break;
337
338 default:
339 break;
340 }
341 }
