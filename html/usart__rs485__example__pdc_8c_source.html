234void USART_Handler(void)
235{
236 uint32_t ul_status;
237
238 /* Read USART status. */
239 ul_status = usart_get_status(BOARD_USART);
240
241 /* Receiving interrupt. */
242 if ((ul_status & US_CSR_ENDRX) && (g_state == RECEIVING)) {
243 /* Indicate receiving finished. */
244 g_state = RECEIVED;
245 usart_disable_interrupt(BOARD_USART, US_IDR_ENDRX);
246 }
247 /* Transmitting interrupt. */
248 else if ((ul_status & US_CSR_ENDTX) && g_state == TRANSMITTING) {
249 /* Transmit continuously. */
250 g_state = TRANSMITTED;
251 usart_disable_interrupt(BOARD_USART, US_IDR_ENDTX);
252 }
253}

333int main(void)
334{
335 static uint8_t uc_sync = SYNC_CHAR;
336 uint32_t time_elapsed = 0;
337 uint32_t ul_i;
338
339 /* Initialize the SAM system. */
340 sysclk_init();
341 board_init();
342
343 /* Configure UART for debug message output. */
344 configure_console();
345
346 /* Output example information. */
347 puts(STRING_HEADER);
348
349 /* Configure USART. */
350 configure_usart();
351
352 /* Get board USART PDC base address and enable receiver and transmitter. */
353 g_p_pdc = usart_get_pdc_base(BOARD_USART);
354 pdc_enable_transfer(g_p_pdc, PERIPH_PTCR_RXTEN | PERIPH_PTCR_TXTEN);
355
356 /* 1ms tick. */
357 configure_systick();
358
359 /* Initialize receiving buffer to distinguish with the sent frame. */
360 memset(g_uc_receive_buffer, 0x0, BUFFER_SIZE);
361
362 /*
363 * Enable transmitter here, and disable receiver first, to avoid receiving
364 * characters sent by itself. It's necessary for half duplex RS485.
365 */
366 usart_enable_tx(BOARD_USART);
367 usart_disable_rx(BOARD_USART);
368
369 /* Send a sync character XON (0x11). */
370 g_st_packet.ul_addr = (uint32_t)&uc_sync;
371 g_st_packet.ul_size = 1;
372 pdc_tx_init(g_p_pdc, &g_st_packet, NULL);
373
374 /* Delay until the line is cleared, an estimated time used. */
375 wait(50);
376
377 /* Read the acknowledgement. */
378 g_st_packet.ul_addr = (uint32_t)&uc_sync;
379 g_st_packet.ul_size = 1;
380 pdc_rx_init(g_p_pdc, &g_st_packet, NULL);
381
382 /* Then enable receiver. */
383 usart_enable_rx(BOARD_USART);
384
385 /* Wait until time out or acknowledgement is received. */
386 time_elapsed = get_tick_count();
387 while (!usart_is_rx_buf_end(BOARD_USART)) {
388 if (get_tick_count() - time_elapsed > TIMEOUT) {
389 break;
390 }
391 }
392
393 /* If acknowledgement received in a short time. */
394 if (usart_is_rx_buf_end(BOARD_USART)) {
395 /* Acknowledgement. */
396 if (uc_sync == ACK_CHAR) {
397 /* Act as transmitter, start transmitting. */
398 g_state = TRANSMITTING;
399 puts("-I- Start transmitting!\r");
400 g_st_packet.ul_addr = (uint32_t)g_uc_transmit_buffer;
401 g_st_packet.ul_size = PDC_BUF_SIZE;
402 pdc_tx_init(g_p_pdc, &g_st_packet, NULL);
403
404 /* Enable transmitting interrupt. */
405 usart_enable_interrupt(BOARD_USART, US_IER_ENDTX);
406 }
407 } else {
408 /* Start receiving, act as receiver. */
409 g_st_packet.ul_addr = (uint32_t)&uc_sync;
410 g_st_packet.ul_size = 1;
411 pdc_rx_init(g_p_pdc, &g_st_packet, NULL);
412 puts("-I- Receiving sync character.\r");
413 while (!usart_is_rx_buf_end(BOARD_USART)) {
414 }
415
416 /* Sync character is received. */
417 if (uc_sync == SYNC_CHAR) {
418 /* SEND XOff as acknowledgement. */
419 uc_sync = ACK_CHAR;
420
421 /*
422 * Delay to prevent the character from being discarded by
423 * transmitter due to responding too soon.
424 */
425 wait(100);
426 pio_set_pin_high(PIN_RE_IDX);
427 g_st_packet.ul_addr = (uint32_t)&uc_sync;
428 g_st_packet.ul_size = 1;
429 pdc_tx_init(g_p_pdc, &g_st_packet, NULL);
430
431 g_state = RECEIVING;
432 puts("-I- Start receiving!\r");
433 g_st_packet.ul_addr = (uint32_t)g_uc_receive_buffer;
434 g_st_packet.ul_size = PDC_BUF_SIZE;
435 pdc_rx_init(g_p_pdc, &g_st_packet, NULL);
436 pio_set_pin_low(PIN_RE_IDX);
437 /* Enable receiving interrupt. */
438 usart_enable_interrupt(BOARD_USART, US_IER_ENDRX);
439 }
440 }
441 while (g_state != RECEIVED) {
442 }
443
444 ul_i = 0;
445 /* Print received frame out. */
446 while ((ul_i < BUFFER_SIZE) && (g_uc_receive_buffer[ul_i] != '\0')) {
447 if (g_uc_transmit_buffer[ul_i] != g_uc_receive_buffer[ul_i]) {
448 puts("-E- Error occurred while receiving!\r");
449 /* Infinite loop here. */
450 while (1) {
451 }
452 }
453 ul_i++;
454 }
455 puts("-I- Received successfully!\r");
456
457 while (1) {
458 }
459}
