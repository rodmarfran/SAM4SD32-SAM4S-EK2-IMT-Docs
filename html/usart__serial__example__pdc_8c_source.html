149void USART_Handler(void)
150{
151 uint32_t ul_status;
152
153 /* Read USART Status. */
154 ul_status = usart_get_status(BOARD_USART);
155
156 if (gs_uc_trans_mode == PDC_TRANSFER) {
157 /* Receive buffer is full. */
158 if (ul_status & US_CSR_RXBUFF) {
159 /* Disable timer. */
160 tc_stop(TC0, 0);
161
162 /* Echo back buffer. */
163 g_st_packet.ul_addr =
164 (uint32_t)gs_puc_buffer[gs_uc_buf_num];
165 g_st_packet.ul_size = gs_ul_size_buffer;
166 g_st_nextpacket.ul_addr =
167 (uint32_t)gs_puc_nextbuffer[gs_uc_buf_num];
168 g_st_nextpacket.ul_size = gs_ul_size_nextbuffer;
169 pdc_tx_init(g_p_pdc, &g_st_packet, &g_st_nextpacket);
170
171 if (g_uc_transend_flag) {
172 gs_ul_size_buffer = BUFFER_SIZE;
173 gs_ul_size_nextbuffer = BUFFER_SIZE;
174 g_uc_transend_flag = 0;
175 }
176
177 gs_uc_buf_num = MAX_BUF_NUM - gs_uc_buf_num;
178
179 /* Restart read on buffer. */
180 g_st_packet.ul_addr =
181 (uint32_t)gs_puc_buffer[gs_uc_buf_num];
182 g_st_packet.ul_size = BUFFER_SIZE;
183 g_st_nextpacket.ul_addr =
184 (uint32_t)gs_puc_nextbuffer[ gs_uc_buf_num];
185 g_st_nextpacket.ul_size = BUFFER_SIZE;
186 pdc_rx_init(g_p_pdc, &g_st_packet, &g_st_nextpacket);
187
188 /* Restart timer. */
189 tc_start(TC0, 0);
190 }
191 } else {
192 /* Transfer without PDC. */
193 if (ul_status & US_CSR_RXRDY) {
194 usart_getchar(BOARD_USART, (uint32_t *)&gs_ul_read_buffer);
195 usart_write(BOARD_USART, gs_ul_read_buffer);
196 }
197 }

204void TC0_Handler(void)
205{
206 uint32_t ul_status;
207 uint32_t ul_byte_total = 0;
208
209 /* Read TC0 Status. */
210 ul_status = tc_get_status(TC0, 0);
211
212 /* RC compare. */
213 if (((ul_status & TC_SR_CPCS) == TC_SR_CPCS) &&
214 (gs_uc_trans_mode == PDC_TRANSFER)) {
215 /* Flush PDC buffer. */
216 ul_byte_total = BUFFER_SIZE - pdc_read_rx_counter(g_p_pdc);
217 if ((ul_byte_total != 0) && (ul_byte_total != BUFFER_SIZE)) {
218 /* Log current size. */
219 g_uc_transend_flag = 1;
220 if (pdc_read_rx_next_counter(g_p_pdc) == 0) {
221 gs_ul_size_buffer = BUFFER_SIZE;
222 gs_ul_size_nextbuffer = ul_byte_total;
223 } else {
224 gs_ul_size_buffer = ul_byte_total;
225 gs_ul_size_nextbuffer = 0;
226 }
227
228 /* Trigger USART Receive Buffer Full Interrupt. */
229 pdc_rx_clear_cnt(g_p_pdc);
230 }
231 }

239static void configure_usart(void)
240{
241 const sam_usart_opt_t usart_console_settings = {
242 BOARD_USART_BAUDRATE,
243 US_MR_CHRL_8_BIT,
244 US_MR_PAR_NO,
245 US_MR_NBSTOP_1_BIT,
246 US_MR_CHMODE_NORMAL,
247 /* This field is only used in IrDA mode. */
248 0
249 };
250
251 /* Enable the peripheral clock in the PMC. */
252 sysclk_enable_peripheral_clock(BOARD_ID_USART);
253
254 /* Configure USART in serial mode. */
255 usart_init_rs232(BOARD_USART, &usart_console_settings,
256#if (SAM4L)
257 sysclk_get_pba_hz());
258#else
259 sysclk_get_peripheral_hz());
260#endif
261
262 /* Disable all the interrupts. */
263 usart_disable_interrupt(BOARD_USART, ALL_INTERRUPT_MASK);
264
265 /* Enable the receiver and transmitter. */
266 usart_enable_tx(BOARD_USART);
267 usart_enable_rx(BOARD_USART);
268
269 /* Configure and enable interrupt of USART. */
270 NVIC_EnableIRQ(USART_IRQn);
271}

277static void configure_tc(void)
278{
279 uint32_t ul_div;
280 uint32_t ul_tcclks;
281 static uint32_t ul_sysclk;
282
283 /* Get system clock. */
284 ul_sysclk = sysclk_get_cpu_hz();
285
286 /* Configure PMC. */
287 pmc_enable_periph_clk(ID_TC0);
288
289 /* Configure TC for a 50Hz frequency and trigger on RC compare. */
290 tc_find_mck_divisor(TC_FREQ, ul_sysclk, &ul_div, &ul_tcclks, ul_sysclk);
291 tc_init(TC0, 0, ul_tcclks | TC_CMR_CPCTRG);
292 tc_write_rc(TC0, 0, (ul_sysclk / ul_div) / TC_FREQ);
293
294 /* Configure and enable interrupt on RC compare. */
295 NVIC_EnableIRQ((IRQn_Type)ID_TC0);
296 tc_enable_interrupt(TC0, 0, TC_IER_CPCS);
297}

355 /* Initialize the SAM system. */
356 sysclk_init();

363 puts(STRING_HEADER);

369 g_p_pdc = usart_get_pdc_base(BOARD_USART);
370 /* Enable receiver and transmitter. */
371 pdc_enable_transfer(g_p_pdc, PERIPH_PTCR_RXTEN | PERIPH_PTCR_TXTEN);
372
373 /* Configure TC. */
374 configure_tc();
375
376 /* Start receiving data and start timer. */
377 g_st_packet.ul_addr = (uint32_t)gs_puc_buffer[gs_uc_buf_num];
378 g_st_packet.ul_size = BUFFER_SIZE;
379 g_st_nextpacket.ul_addr = (uint32_t)gs_puc_nextbuffer[gs_uc_buf_num];
380 g_st_nextpacket.ul_size = BUFFER_SIZE;
381 pdc_rx_init(g_p_pdc, &g_st_packet, &g_st_nextpacket);
382
383 puts("-- Start to echo serial inputs -- \r\n"
384 "-I- Default Transfer with PDC \r\n"
385 "-I- Press 's' to switch transfer mode \r");
386 gs_uc_trans_mode = PDC_TRANSFER;
387
388 usart_disable_interrupt(BOARD_USART, US_IDR_RXRDY);
389 usart_enable_interrupt(BOARD_USART, US_IER_RXBUFF);
390
391 tc_start(TC0, 0);
392
393 while (1) {
394 uc_char = 0;
395 uc_flag = uart_read(CONSOLE_UART, &uc_char);
396 if (!uc_flag) {
397 switch (uc_char) {
398 case 's':
399 case 'S':
400 if (gs_uc_trans_mode == PDC_TRANSFER) {
401 /* Transfer to no PDC communication mode. */
402 /* Disable PDC controller. */
403 pdc_disable_transfer(g_p_pdc,
404 PERIPH_PTCR_RXTDIS | PERIPH_PTCR_TXTDIS);
405 /* Disable the RXBUFF interrupt. */
406 usart_disable_interrupt(BOARD_USART, US_IDR_RXBUFF);
407
408 /* Clear USART controller. */
409 usart_clear();
410
411 /* Enable the RXRDY interrupt. */
412 usart_enable_interrupt(BOARD_USART, US_IER_RXRDY);
413 gs_uc_trans_mode = BYTE_TRANSFER;
414
415 puts("-I- Transfer without PDC \r");
416 } else if (gs_uc_trans_mode == BYTE_TRANSFER) {
417 pdc_enable_transfer(g_p_pdc,
418 PERIPH_PTCR_RXTEN | PERIPH_PTCR_TXTEN);
419 /* Clear USART controller. */
420 usart_clear();
421
422 /* Reset pdc current buffer size. */
423 gs_ul_size_buffer = BUFFER_SIZE;
424 gs_ul_size_nextbuffer = BUFFER_SIZE;
425 gs_uc_buf_num = 0;
426
427 /* Start receiving data. */
428 g_st_packet.ul_addr =
429 (uint32_t)gs_puc_buffer[gs_uc_buf_num];
430 g_st_packet.ul_size = BUFFER_SIZE;
431 g_st_nextpacket.ul_addr =
432 (uint32_t)gs_puc_nextbuffer[gs_uc_buf_num];
433 g_st_nextpacket.ul_size = BUFFER_SIZE;
434 pdc_rx_init(g_p_pdc, &g_st_packet, &g_st_nextpacket);
435
436 /* Transfer to PDC communication mode, disable RXRDY interrupt and enable RXBUFF interrupt. */
437 usart_disable_interrupt(BOARD_USART, US_IER_RXRDY);
438 usart_enable_interrupt(BOARD_USART, US_IER_RXBUFF);
439
440 gs_uc_trans_mode = PDC_TRANSFER;
441 puts("-I- Transfer with PDC \r");
442 }
443
444 break;
445
446 case 'm':
447 case 'M':
448 display_main_menu();

450

453 }
