186void USART_Handler(void)
187{
188 uint32_t ul_status;
189
190 /* Read USART Status. */
191 ul_status = usart_get_status(BOARD_USART);
192
193 /* Receive buffer is full. */
194 if ((ul_status & US_CSR_RXBUFF) && (g_uc_state == STATE_READ)) {
195 g_ul_recv_done = true;
196 usart_disable_interrupt(BOARD_USART, US_IDR_RXBUFF);
197 }
198 if ((ul_status & US_CSR_TXBUFE) && (g_uc_state == STATE_WRITE)) {
199 g_ul_sent_done = true;
200 usart_disable_interrupt(BOARD_USART, US_IDR_TXBUFE);
201 }
202}

232static void configure_usart(uint32_t ul_ismaster, uint32_t ul_baudrate)
233{
234 sam_usart_opt_t usart_console_settings = {
235 0,
236 US_MR_CHRL_8_BIT,
237 US_MR_PAR_NO,
238 US_MR_NBSTOP_1_BIT,
239 US_MR_CHMODE_NORMAL,
240 /* This field is only used in IrDA mode. */
241 0
242 };
243
244 usart_console_settings.baudrate = ul_baudrate;
245
246#if (SAMG55)
247 /* Enable the peripheral and set USART mode. */
248 flexcom_enable(BOARD_FLEXCOM);
249 flexcom_set_opmode(BOARD_FLEXCOM, FLEXCOM_USART);
250#else
251 /* Enable the peripheral clock in the PMC. */
252 sysclk_enable_peripheral_clock(BOARD_ID_USART);
253#endif
254
255 /* Configure USART in SYNC. master or slave mode. */
256 if (ul_ismaster) {
257 usart_init_sync_master(BOARD_USART, &usart_console_settings, sysclk_get_peripheral_hz());
258 } else {
259 usart_init_sync_slave(BOARD_USART, &usart_console_settings);
260 }
261
262 /* Disable all the interrupts. */
263 usart_disable_interrupt(BOARD_USART, ALL_INTERRUPT_MASK);
264
265 /* Enable TX & RX function. */
266 usart_enable_tx(BOARD_USART);
267 usart_enable_rx(BOARD_USART);
268
269 /* Configure and enable interrupt of USART. */
270 NVIC_EnableIRQ(USART_IRQn);
271}

307int main(void)
308{
309 uint8_t uc_char;
310
311 /* Initialize the SAM system. */
312 sysclk_init();
313 board_init();
314
315 /* Configure UART for debug message output. */
316 configure_console();
317
318 /* Output example information. */
319 puts(STRING_HEADER);
320
321 /* Display main menu. */
322 display_main_menu();
323
324 /* Configure USART. */
325 configure_usart(SYNC_MASTER, g_ul_freq[g_uc_freq_idx]);
326
327 /* Get board USART PDC base address and enable receiver and transmitter. */
328 g_p_pdc = usart_get_pdc_base(BOARD_USART);
329 pdc_enable_transfer(g_p_pdc, PERIPH_PTCR_RXTEN | PERIPH_PTCR_TXTEN);
330
331 g_uc_transfer_mode = SYNC_MASTER;
332
333 g_uc_state = STATE_WRITE;
334
335 puts("-- USART in MASTER mode --\r");
336
337 while (1) {
338 uc_char = 0;
339 scanf("%c", (char *)&uc_char);
340 switch (uc_char) {
341 case '0':
342 case '1':
343 case '2':
344 case '3':
345 g_uc_freq_idx = uc_char - '0';
346 printf("-- The clock freq is: %luHz.\r\n",
347 (unsigned long)g_ul_freq[g_uc_freq_idx]);
348 configure_usart(SYNC_MASTER, g_ul_freq[g_uc_freq_idx]);
349 break;
350 case 'i':
351 case 'I':
352 if (g_uc_transfer_mode == SYNC_MASTER) {
353 printf("-- USART is MASTER at %luHz.\r\n",
354 (unsigned long)g_ul_freq[g_uc_freq_idx]);
355 } else {
356 puts("-- USART is SLAVE \r");
357 }
358 break;
359 case 's':
360 case 'S':
361 if (g_uc_transfer_mode == SYNC_MASTER) {
362 g_uc_transfer_mode = SYNC_SLAVE;
363 configure_usart(SYNC_SLAVE, g_ul_freq[g_uc_freq_idx]);
364 puts("-- USART in SLAVE mode --\r");
365 } else {
366 if (g_uc_transfer_mode == SYNC_SLAVE) {
367 g_uc_transfer_mode = SYNC_MASTER;
368 configure_usart(SYNC_MASTER, g_ul_freq[g_uc_freq_idx]);
369 puts("-- USART in MASTER mode --\r");
370 }
371 }
372 break;
373 case 'w':
374 case 'W':
375 g_uc_state = STATE_WRITE;
376 g_st_packet.ul_addr = (uint32_t)tran_buff;
377 g_st_packet.ul_size = BUFFER_SIZE;
378 pdc_tx_init(g_p_pdc, &g_st_packet, NULL);
379 usart_enable_interrupt(BOARD_USART, US_IER_TXBUFE);
380 while (!g_ul_sent_done) {
381 }
382 if (g_ul_sent_done) {
383 printf("-- %s sent done --\r\n",
384 g_uc_transfer_mode ? "MASTER" :
385 "SLAVE");
386 }
387 break;
388 case 'r':
389 case 'R':
390 g_uc_state = STATE_READ;
391 if (g_uc_transfer_mode == SYNC_MASTER) {
392 puts("----USART MASTER Read----\r");
393 } else {
394 puts("----USART SLAVE Read----\r");
395 }
396 g_st_packet.ul_addr = (uint32_t)g_c_recv_buff;
397 g_st_packet.ul_size = BUFFER_SIZE;
398 pdc_rx_init(g_p_pdc, &g_st_packet, NULL);
399 usart_enable_interrupt(BOARD_USART, US_IER_RXBUFF);
400 while (!g_ul_recv_done) {
401 }
402 if (g_ul_recv_done) {
403 if (strncmp((char*)g_c_recv_buff, tran_buff, BUFFER_SIZE)) {
404 puts(" -F-: Failed!\r");
405 } else {
406 /* successfully received */
407 dump_info((char*)g_c_recv_buff, BUFFER_SIZE);
408 }
409 puts("----END of read----\r");
410 memset(g_c_recv_buff, 0, sizeof(g_c_recv_buff));
411 g_ul_recv_done = false;
412 }
413 break;
414 case 'm':
415 case 'M':
416 display_main_menu();
417 break;
418 default:
419 break;
420 }
421 }
